## Description

## Example

### Open a drop-in co-op practice room

Open a four-seat practice room so friends can join after training starts. Use a room with four spawn points. [Set up the training room](/multiplayer-quickstart)

```gdscript
extends Node

const PRACTICE_ARENA: StringName = &"training_room"
const PRACTICE_CAPACITY: int = 4
const PRACTICE_PORT: int = 7017
const LOCAL_ADDRESS: String = "127.0.0.1"

@export var arena_scene: PackedScene
@export var player_scene: PackedScene
@export var host_practice: Button
@export var status: Label

var _game: GameRuntime


func _ready() -> void:
    host_practice.pressed.connect(_host_practice)


func _host_practice() -> void:
    if arena_scene == null or player_scene == null:
        status.text = "Assign the practice arena and player scenes."
        return

    host_practice.disabled = true
    status.text = "Opening the practice room..."
    var rules: MultiplayerRules = (
        MultiplayerRules.listen_coop()
        .with_max_players(PRACTICE_CAPACITY)
        .allow_join_in_progress(true)
        .on_port(PRACTICE_PORT)
    )
    var definition: GameDefinition = (
        GameRuntime.configure()
        .with_starting_world(PRACTICE_ARENA, arena_scene)
        .with_player(player_scene)
        .with_multiplayer(rules)
    )
    _game = definition.boot(self)
    if _game == null:
        status.text = "Could not create the practice room."
        host_practice.disabled = false
        return

    var result: Dictionary = await _game.host_game(LOCAL_ADDRESS).wait()
    if not result.ok:
        status.text = "Could not host: %s" % result.message
        _game.queue_free()
        await _game.tree_exited
        _game = null
        host_practice.disabled = false
        return

    host_practice.hide()
    status.text = "Practice is open to %d players, including you." % PRACTICE_CAPACITY
```
