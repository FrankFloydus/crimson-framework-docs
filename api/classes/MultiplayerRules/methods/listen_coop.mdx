## Example

```gdscript
extends Node

const COOP_PORT: int = 7017
const TOTAL_SEATS: int = 4

@export var game_definition: GameDefinition

# Connect the Host button. Assign a player scene and a world with at least four player spawn points.
func host_coop() -> void:
    if game_definition == null:
        push_error("Assign a game definition.")
        return
    var rules: MultiplayerRules = MultiplayerRules.listen_coop()
    rules = rules.on_port(COOP_PORT).with_max_players(TOTAL_SEATS)
    var definition: GameDefinition = game_definition.with_multiplayer(rules)
    var game: GameRuntime = definition.boot(self)
    if game == null:
        return
    var hosted: Dictionary = await game.host_game().wait()
    if not hosted.ok:
        push_error(hosted.message)
        game.queue_free()
        return
    print("The co-op world is ready and accepts late joins on UDP port ", rules.port)
```
