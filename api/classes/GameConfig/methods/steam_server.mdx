## Example

### Publish a dedicated Steam arena

Start a headless Steam arena and log its locator. Use your published app's product metadata and Steam server launch context.

```gdscript
extends Node

const ARENA_ID: StringName = &"arena"
const GUEST_SEATS: int = 4
const GAME_PORT: int = 27015
const QUERY_PORT: int = 27016
const SERVER_NAME: String = "Cargo Yard Arena"
const BUILD_VERSION: String = "1.0.0.0"
const GAME_NAME: String = "Cargo Yard"
const INSTALL_DIR: String = "Cargo Yard"
const EXIT_FAILURE: int = 1

@export var arena: PackedScene
@export var player: PackedScene

func _ready() -> void:
    var steam_settings: SteamServer = SteamServer.configure(SERVER_NAME, GAME_PORT, QUERY_PORT)
    steam_settings = steam_settings.with_product_metadata(BUILD_VERSION, GAME_NAME, INSTALL_DIR)
    steam_settings = steam_settings.advertise_on(SteamServer.INTERNET)
    var setup: GameConfig = Game.configure(arena, player).world_id(ARENA_ID)
    setup = setup.steam_server(steam_settings, GUEST_SEATS)
    var server: Game = setup.boot(self)
    var started: GameResult = await server.serve()
    if not started.ok:
        push_error(started.message)
        get_tree().quit(EXIT_FAILURE)
        return
    print("Steam arena locator: ", started.listing_id)
```
