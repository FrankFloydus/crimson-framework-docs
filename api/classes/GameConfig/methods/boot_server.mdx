## Description

## Example

### Boot the cargo yard from a headless launcher

Launch the headless scene with `-- --crimson-role=server`. The server opens four guest seats and reports when the cargo yard is ready.

```gdscript
extends Node

const CARGO_YARD: StringName = &"cargo_yard"
const SERVER_PORT: int = 24830
const GUEST_SEATS: int = 4

@export var yard: PackedScene
@export var player: PackedScene

func _ready() -> void:
    var setup: GameConfig = Game.configure(yard, player).world_id(CARGO_YARD)
    setup = setup.lan(GUEST_SEATS).on_port(SERVER_PORT)
    var server: Game = setup.boot_server(self)
    server.world_ready.connect(_on_world_ready)

func _on_world_ready(world_id: StringName, _epoch: int) -> void:
    print("Dedicated world is ready: ", world_id)
```
