extends Node

func run_get_value(
    steam_item_definition_result: SteamItemDefinitionResult,
    display_name_label: Label
) -> void:

        return
    if not steam_item_definition_result.ok:

        return
    var result: SteamItemDefinition = steam_item_definition_result.get_value()
