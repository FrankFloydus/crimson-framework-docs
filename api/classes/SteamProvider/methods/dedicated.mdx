## Signature

```text
dedicated(settings: SteamServer) -> SteamProvider
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `settings` | [`SteamServer`](/api/classes/SteamServer) | — |

## Return value

[`SteamProvider`](/api/classes/SteamProvider)

## Description

The game remains playerless and does not acquire a signed-in Steam client. Null settings return null.

## Example

```gdscript
extends Node

# Set the AppID and product metadata for the Steam title you administer.
const STEAM_APP_ID: int = 480
const SERVER_NAME: String = "Harbor Training Server"
const GAME_PORT: int = 27015
const QUERY_PORT: int = 27016
const LOBBY_NAMESPACE: String = "crimson-harbor-dedicated"
const TOTAL_SEATS: int = 2
const PRODUCT_VERSION: String = "1.0.0.0"
const PRODUCT_DESCRIPTION: String = "Spacewar"
const PRODUCT_DIRECTORY: String = "Spacewar"

# The world needs at least two player spawn points for this capacity.
@export var game_definition: GameDefinition

# Attach this script to the root of the dedicated process's scene.
func _ready() -> void:
    if game_definition == null:
        push_error("Assign the required dedicated server settings in the Inspector.")
        return

    var settings: SteamServer = SteamServer.configure(SERVER_NAME, GAME_PORT, QUERY_PORT)
    settings = settings.with_product_metadata(PRODUCT_VERSION, PRODUCT_DESCRIPTION, PRODUCT_DIRECTORY)
    var provider: SteamProvider = SteamProvider.configure().for_app(STEAM_APP_ID)
    provider = provider.with_lobby_namespace(LOBBY_NAMESPACE)
    # This branch opens a SteamGameServer session for the playerless world.
    provider = provider.dedicated(settings)
    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    var game: GameRuntime = game_definition.with_provider(provider).with_multiplayer(rules).boot(self)
    if game == null:
        push_error("Could not attach the dedicated runtime.")
        return

    var started: Dictionary = await game.start_dedicated_online().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("Dedicated server locator: ", game.get_online_listing())
```
