## Example

```gdscript
extends Node

# Replace Spacewar's test AppID with the one you use to launch your game.
const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"

@export var game_definition: GameDefinition
@export var status_label: Label

var _provider: SteamProvider
var _game: GameRuntime

func _ready() -> void:
    if game_definition == null:
        push_error("Assign a game definition with the world and player scenes.")
        return
    # Keep this branch exclusive to this game attachment.
    _provider = SteamProvider.configure().for_app(STEAM_APP_ID)
    _provider = _provider.with_lobby_namespace(LOBBY_NAMESPACE)
    if status_label == null:
        return
    _game = game_definition.with_provider(_provider).boot(self)
    if _game == null:
        push_error("Could not attach the game runtime.")
        return
    if not _provider.is_ready():
        var diagnostic: String = _provider.get_initialization_error()
        status_label.text = diagnostic if not diagnostic.is_empty() else "Steam is unavailable."
        return
    status_label.text = "Connected to the local Steam client."
```
