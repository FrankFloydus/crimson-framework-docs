## Example

```gdscript
extends Node

# Replace Spacewar's test AppID with the one you use to launch your game.
const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const TOTAL_SEATS: int = 2

# The world needs at least two player spawn points for this capacity.
@export var game_definition: GameDefinition
@export var invitee_account_id: String

var _provider: SteamProvider
var _game: GameRuntime

func _ready() -> void:
    if game_definition == null:
        push_error("Assign a game definition with the world and player scenes.")
        return
    # Keep this branch exclusive to this game attachment.
    _provider = SteamProvider.configure().for_app(STEAM_APP_ID)
    _provider = _provider.with_lobby_namespace(LOBBY_NAMESPACE)
    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    _game = game_definition.with_provider(_provider).with_multiplayer(rules).boot(self)
    if _game == null:
        push_error("Could not attach the game runtime.")
        return
    if not _provider.is_ready():
        push_warning(_provider.get_initialization_error())
        return
    var hosted: Dictionary = await _game.host_online().wait()
    if not hosted.ok:
        push_warning(hosted.message)
        return
    _hosted = true
    print("Private lobby ready: ", _game.get_online_listing())

var _hosted: bool = false

# Call this only when the host confirms the friend selected by your UI.
func invite_selected_friend() -> void:
    if not _hosted or not is_instance_valid(_game) or _game.get_online_listing().is_empty():
        return
    var steam_id: String = invitee_account_id.strip_edges()
    if steam_id.is_empty():
        return
    var sent: Error = _provider.invite_player(steam_id)
    if sent != OK:
        push_warning(error_string(sent))
    else:
        print("Invite submitted to Steam.")
```
