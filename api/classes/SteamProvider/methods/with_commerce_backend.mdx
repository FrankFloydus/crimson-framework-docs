## Signature

```text
with_commerce_backend(backend: SteamCommerceBackend) -> SteamProvider
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `backend` | [`SteamCommerceBackend`](/api/classes/SteamCommerceBackend) | — |

## Return value

[`SteamProvider`](/api/classes/SteamProvider)

## Description

The adapter must use the game's authenticated purchasing backend. Pass null for a provider without commerce.

## Example

```gdscript
extends Node

# Assign a SteamCommerceBackend subclass that sends authenticated shop requests.
@export var game_definition: GameDefinition
@export var steam_app_id: int

var _game: GameRuntime

# Call this with your authenticated shop adapter after this node enters the scene tree.
func attach_shop_backend(commerce_backend: SteamCommerceBackend) -> void:
    if is_instance_valid(_game):
        return
    if game_definition == null or commerce_backend == null or steam_app_id <= 0:
        push_error("Assign the game definition and Steam AppID, then pass a commerce adapter.")
        return
    var provider: SteamProvider = SteamProvider.configure().for_app(steam_app_id)
    if provider == null:
        push_error("Invalid Steam AppID.")
        return
    provider = provider.with_commerce_backend(commerce_backend)
    _game = game_definition.with_provider(provider).boot(self)

# Connect the Shop button to this method after attaching the game.
func refresh_store() -> void:
    if not is_instance_valid(_game):
        return
    var steam: SteamScope = SteamServices.for_game(_game)
    if steam == null or not steam.is_ready():
        push_warning("Steam is unavailable for this game.")
        return
    var purchases: SteamPurchases = steam.purchases()
    if purchases == null:
        push_warning("The shop service is unavailable.")
        return
    var catalog: SteamPurchaseCatalogResult = await purchases.catalog()
    if not catalog.ok:
        push_warning(catalog.message)
        return
    print("Offers for the signed-in player: ", catalog.value.entries.size())
```
