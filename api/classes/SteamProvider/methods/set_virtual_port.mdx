## Example

```gdscript
extends Node

# Use the Spacewar test AppID while learning. Replace it with your own AppID.
const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const TOTAL_SEATS: int = 2
const EVENT_VIRTUAL_PORT: int = 17

# The world needs at least two player spawn points for this capacity.
@export var game_definition: GameDefinition

var _game: GameRuntime

# Connect the Host button to this method.
func host_event_match() -> void:
    if is_instance_valid(_game):
        return
    if game_definition == null:
        push_error("Assign a game definition with the world and player scenes.")
        return

    var provider: SteamProvider = SteamProvider.configure().for_app(STEAM_APP_ID)
    provider = provider.with_lobby_namespace(LOBBY_NAMESPACE)
    provider.set_virtual_port(EVENT_VIRTUAL_PORT)
    if provider == null:
        push_error("The Steam settings are invalid.")
        return

    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    _game = game_definition.with_provider(provider).with_multiplayer(rules).boot(self)
    if _game == null:
        push_error("Could not attach the game runtime.")
        return

    var hosted: Dictionary = await _game.host_online().wait()
    if not hosted.ok:
        push_error(hosted.message)
        _game.queue_free()
        _game = null
        return
    print("The event is ready. Guests must use Steam virtual port ", provider.get_virtual_port())
```
