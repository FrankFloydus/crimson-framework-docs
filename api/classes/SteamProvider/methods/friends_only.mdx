## Example

```gdscript
extends Node

# Use the Spacewar test AppID while learning. Replace it with your own AppID.
const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const TOTAL_SEATS: int = 2
const WORLD_ID: StringName = &"harbor"

# The world needs at least two player spawn points for this capacity.
@export var world_scene: PackedScene
@export var player_scene: PackedScene

var _game: GameRuntime

# Connect the Host button to this method.
# The host's Steam friends can join this lobby without a public search.
func host_for_steam_friends() -> void:
    if is_instance_valid(_game):
        return
    if world_scene == null or player_scene == null:
        push_error("Assign the harbor scene and player scene in the Inspector.")
        return

    var provider: SteamProvider = SteamProvider.configure().for_app(STEAM_APP_ID)
    provider = provider.with_lobby_namespace(LOBBY_NAMESPACE).friends_only()

    var definition: GameDefinition = GameDefinition.configure()
    definition = definition.with_starting_world(WORLD_ID, world_scene)
    definition = definition.with_player(player_scene)
    if provider == null:
        push_error("The Steam settings are invalid.")
        return

    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    _game = definition.with_provider(provider).with_multiplayer(rules).boot(self)
    if _game == null:
        push_error("Could not attach the game runtime.")
        return

    var hosted: Dictionary = await _game.host_online().wait()
    if not hosted.ok:
        push_error(hosted.message)
        _game.queue_free()
        _game = null
        return
    print("Friends and invitees can join the harbor. Lobby: ", _game.get_online_listing())
```
