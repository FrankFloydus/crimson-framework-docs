## Signature

```text
with_message_channels(channels: StringName[]) -> SteamProvider
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `channels` | `StringName[]` | — |

## Return value

[`SteamProvider`](/api/classes/SteamProvider)

## Description

Supply at most 16 distinct channel names of up to 32 ASCII characters, or an empty list to disable Game Messages. Invalid names return null.

## Example

```gdscript
extends Node

# Use the Spacewar test AppID while learning. Replace it with your own AppID.
const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const TOTAL_SEATS: int = 2
const SQUAD_CHANNEL: StringName = &"squad"

# The world needs at least two player spawn points for this capacity.
@export var game_definition: GameDefinition

var _game: GameRuntime

# Connect the Host button to this method.
func host_squad_game() -> void:
    if is_instance_valid(_game):
        return
    if game_definition == null:
        push_error("Assign a game definition with the world and player scenes.")
        return

    var provider: SteamProvider = SteamProvider.configure().for_app(STEAM_APP_ID)
    provider = provider.with_lobby_namespace(LOBBY_NAMESPACE).friends_only()
    var channels: Array[StringName] = [SQUAD_CHANNEL]
    provider = provider.with_message_channels(channels)
    if provider == null:
        push_error("The Steam settings are invalid.")
        return

    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    _game = game_definition.with_provider(provider).with_multiplayer(rules).boot(self)
    if _game == null:
        push_error("Could not attach the game runtime.")
        return

    var hosted: Dictionary = await _game.host_online().wait()
    if not hosted.ok:
        push_error(hosted.message)
        _game.queue_free()
        _game = null
        return
    print("Squad lobby ready. Send squad packets only after the target peer is world-ready.")

# Call this with the current peer ID from the game's admitted roster.
func send_squad_update(peer_id: int, payload: PackedByteArray) -> void:
    if not is_instance_valid(_game):
        return
    var steam: SteamScope = SteamServices.for_game(_game)
    if steam == null:
        return
    var messages: SteamMessages = steam.messages()
    if messages == null:
        return
    var peer: SteamMessagePeerResult = messages.for_peer(peer_id)
    if not peer.ok:
        push_warning(peer.message)
        return
    var sent: Error = peer.value.send(SQUAD_CHANNEL, payload)
    if sent != OK:
        push_warning(error_string(sent))
```
