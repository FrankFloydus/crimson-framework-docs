## Description

Private requires an invitation. Friends Only permits Steam friends and invitees. Public appears in compatible lobby search. Invisible also appears in search and hides the lobby from friends visibility.

## Example

```gdscript
extends Node

const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const TOTAL_SEATS: int = 2

# The world needs at least two player spawn points for this capacity.
@export var game_definition: GameDefinition
@export var status_label: Label

# Apply the Steam settings before attaching a new game.
func _ready() -> void:
    if game_definition == null or status_label == null:
        push_error("Assign the game definition and its status label.")
        return
    var provider: SteamProvider = SteamProvider.configure()
    provider.app_id = STEAM_APP_ID
    provider.lobby_namespace = LOBBY_NAMESPACE
    provider.lobby_visibility = SteamProvider.FRIENDS_ONLY
    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    var game: GameRuntime = game_definition.with_provider(provider).with_multiplayer(rules).boot(self)
    if game == null:
        status_label.text = "Could not attach the game."
        return
    var hosted: Dictionary = await game.host_online().wait()
    status_label.text = "Friends lobby open" if hosted.ok else hosted.message
```
