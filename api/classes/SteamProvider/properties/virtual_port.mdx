## Description

Zero is the standard Steam P2P port. This value is independent of the direct UDP port.

## Example

```gdscript
extends Node

const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const EVENT_VIRTUAL_PORT: int = 17

@export var game_definition: GameDefinition
@export var status_label: Label

# Apply the Steam settings before attaching a new game.
func _ready() -> void:
    if game_definition == null or status_label == null:
        push_error("Assign the game definition and its status label.")
        return
    var provider: SteamProvider = SteamProvider.configure()
    provider.app_id = STEAM_APP_ID
    provider.lobby_namespace = LOBBY_NAMESPACE
    provider.virtual_port = EVENT_VIRTUAL_PORT
    var game: GameRuntime = game_definition.with_provider(provider).boot(self)
    if game == null:
        status_label.text = "Could not attach the game."
        return
    status_label.text = "Joining clients need virtual port %d." % provider.virtual_port
```
