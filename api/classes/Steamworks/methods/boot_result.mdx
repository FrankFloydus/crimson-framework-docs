## Signature

```text
boot_result() -> SteamResult
```

## Return value

[`SteamResult`](/api/classes/SteamResult)

## Description

Use `is_ready()` when a view needs to observe current availability.

## Example

### Enable online play after Steam connects

The title screen enables Online Play when Steam connects and leaves Solo Play available during an outage. Launch the game through Steam.

```gdscript
extends Control

const STEAM_CHECK_SECONDS: float = 1.0

@export var online_play: Button
@export var status: Label

var _steam: Steamworks

func _ready() -> void:
    online_play.disabled = true
    _steam = Steamworks.boot(self)
    var booted: SteamResult = _steam.boot_result()
    if not booted.ok:
        status.text = "Steam unavailable. You can still play solo."
        push_warning(booted.message)
        return

    var refresh: Timer = Timer.new()
    refresh.wait_time = STEAM_CHECK_SECONDS
    refresh.timeout.connect(_refresh_online_status)
    add_child(refresh)
    refresh.start()
    _refresh_online_status()

func _refresh_online_status() -> void:
    var online: bool = _steam.is_ready() and _steam.is_online()
    online_play.disabled = not online
    status.text = "Online play available." if online else "Steam offline. Solo play available."
```
