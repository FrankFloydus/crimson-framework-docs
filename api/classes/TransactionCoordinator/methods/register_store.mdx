## Signature

```text
register_store(store_id: String, initial_state: Dictionary, prepare: Callable, validate_restore: Callable = Callable()) -> Error
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `store_id` | `String` | — |
| `initial_state` | `Dictionary` | — |
| `prepare` | `Callable` | — |
| `validate_restore` | `Callable` | `Callable()` |

## Return value

`Error`

## Description

Preparation returns a dictionary with boolean `ok`, a dictionary `state` on success, and an optional `message`. The optional `validate\_restore(candidate, context)` callback returns boolean `ok` in a dictionary and can include `message`. Without it, restore calls the preparer with an empty operation array.

## Example

```gdscript
extends Node

const WALLET_STORE: String = "player_wallet"
const INITIAL_COINS: int = 10
const REWARD_COINS: int = 5

func _ready() -> void:
    var coordinator: TransactionCoordinator = TransactionCoordinator.new()
    var initial_wallet: Dictionary = {"coins": INITIAL_COINS}
    var registered: Error = coordinator.register_store(
        WALLET_STORE, initial_wallet, _prepare_wallet
    )
    if registered != OK:
        push_error(error_string(registered))
        return

    var reward: GameTransaction = coordinator.begin()
    var revision: int = coordinator.get_store_revision(WALLET_STORE)
    var queued: Error = reward.add_change(WALLET_STORE, {"add": REWARD_COINS}, revision)
    if queued != OK:
        push_warning(error_string(queued))
        return

    var committed: Dictionary = coordinator.commit(reward)
    if not committed.get("ok", false):
        push_warning(String(committed.get("message", "Reward rejected.")))
        return

    var wallet: Dictionary = coordinator.read_store(WALLET_STORE)
    print("Show this wallet balance in the HUD: ", wallet["coins"])

func _prepare_wallet(state: Dictionary, operations: Array, _context: Dictionary) -> Dictionary:
    for operation: Dictionary in operations:
        var amount: int = int(operation.get("add", 0))
        if amount <= 0:
            return {"ok": false, "message": "A reward must be positive."}
        state["coins"] = int(state["coins"]) + amount
    return {"ok": true, "state": state}
```
