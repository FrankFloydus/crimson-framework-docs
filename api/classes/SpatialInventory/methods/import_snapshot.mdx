## Signature

```text
import_snapshot(snapshot: Dictionary) -> Error
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `snapshot` | `Dictionary` | — |

## Return value

`Error`

## Description

An invalid snapshot preserves the current grid.

## Example

```gdscript
extends Node

const BAG_SIZE: Vector2i = Vector2i(5, 4)
const MAP_ITEM: StringName = &"folded_map_01"
const MAP_FOOTPRINT: Vector2i = Vector2i(2, 1)
const MAP_CELL: Vector2i = Vector2i(1, 2)

func _ready() -> void:
    var original: SpatialInventory = SpatialInventory.new()
    var configured: Error = original.configure(BAG_SIZE)
    if configured != OK:
        push_error(error_string(configured))
        return
    var placed: Error = original.try_place(MAP_ITEM, MAP_FOOTPRINT, MAP_CELL)
    if placed != OK:
        push_error(error_string(placed))
        return

    # Store this placement dictionary with your checkpoint.
    var checkpoint_grid: Dictionary = original.export_snapshot()

    # Restore into a fresh grid before rebuilding the inventory UI.
    var restored: SpatialInventory = SpatialInventory.new()
    var imported: Error = restored.import_snapshot(checkpoint_grid)
    if imported != OK:
        push_warning("Saved inventory could not be loaded. " + error_string(imported))
        return

    var item_at_cell: StringName = restored.get_item_at(MAP_CELL)
    print("Redraw cell ", MAP_CELL, " with item ", item_at_cell)
```
