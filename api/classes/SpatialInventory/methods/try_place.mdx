## Example

```gdscript
const MEDKIT_SIZE: Vector2i = Vector2i(2, 1)
const NO_FIT: Vector2i = Vector2i(-1, -1)

func place_medkit(inventory: SpatialInventory, item_id: StringName) -> Error:
    var cell: Vector2i = inventory.find_first_fit(MEDKIT_SIZE)
    if cell == NO_FIT:
        return ERR_CANT_ACQUIRE_RESOURCE
    var placed: Error = inventory.try_place(item_id, MEDKIT_SIZE, cell)
    if placed != OK:
        push_warning(error_string(placed))
    else:
        print("Medkit placed at ", cell)
    return placed
```
