## Example

### Pack a starting kit

Press **Pack supplies** to load a medkit and radio into a bag. The output shows the first available placement for each item. [Build a visible backpack](/inventory/quickstart)

```gdscript
extends Button

const BAG_SIZE: Vector2i = Vector2i(4, 3)
const MEDKIT_ID: StringName = &"medkit_17"
const MEDKIT_SIZE: Vector2i = Vector2i(2, 1)
const RADIO_ID: StringName = &"radio_02"
const RADIO_SIZE: Vector2i = Vector2i(1, 2)
const RADIO_CELL: Vector2i = Vector2i(2, 1)

var _bag: SpatialInventory


func _ready() -> void:
    pressed.connect(_pack_supplies)


func _pack_supplies() -> void:
    var bag: SpatialInventory = Inventory.grid(BAG_SIZE)
    if bag == null:
        push_error("Invalid supply bag dimensions.")
        return
    var error: Error = bag.add(MEDKIT_ID, MEDKIT_SIZE)
    if error == OK:
        error = bag.add(RADIO_ID, RADIO_SIZE)
    if error != OK:
        push_error("Could not pack the supplies: %s" % error_string(error))
        return

    _bag = bag
    var placements: Array[InventoryItemView] = _bag.items()
    for item: InventoryItemView in placements:
        print("%s at %s, footprint %s" % [item.id, item.origin, item.footprint])

    var inspected: InventoryItemView = _bag.item_at(RADIO_CELL)
    if inspected != null:
        print("Cell %s contains %s." % [RADIO_CELL, inspected.id])
    text = "Supplies packed"
    disabled = true
```
