func run_result(
    steam_item_definition_operation: SteamItemDefinitionOperation,
    display_name_label: Label
) -> void:
    var result: SteamItemDefinitionResult = steam_item_definition_operation.result()
    if result == null:

        return
    var payload: SteamItemDefinition = result.value
