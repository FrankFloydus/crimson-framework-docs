## Return value

[`OnlineDirectory`](/api/classes/OnlineDirectory)

## Description

Configure an authenticator on the session as well to use authenticated online host and join operations.

## Example

```gdscript
extends GameServiceProvider

# Assign a scripted directory resource with its backend configuration.
@export var directory_template: OnlineDirectory

func _create_online_directory() -> OnlineDirectory:
    if directory_template == null:
        return null
    # Give each game a separate directory with the same exported settings.
    var directory: OnlineDirectory = directory_template.duplicate() as OnlineDirectory
    if directory == null:
        push_error("Could not create the game's online directory.")
    return directory
```
