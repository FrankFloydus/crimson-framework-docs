## Example

```gdscript
extends Node

func run_on_map(
    steam_lobbies: SteamLobbies,
    world_id: StringName,
    results_list: ItemList
) -> void:
    var steam_lobby_query: SteamLobbyQuery = steam_lobbies.query()
    if steam_lobby_query == null:
        return
    var result: SteamLobbyQuery = steam_lobby_query.on_map(world_id)
    if result == null:
        return
    var completion: SteamListingsResult = await result.fetch()
    if completion == null:
        return
    if not completion.ok:
        push_warning(completion.message)
        return
    var payload: Array[SteamListing] = completion.value
    results_list.clear()
    for entry: SteamListing in payload:
        results_list.add_item(str(entry.name))
```
