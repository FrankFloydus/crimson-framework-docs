## Signature

```text
fetch_operation() -> SteamListingsOperation
```

## Return value

[`SteamListingsOperation`](/api/classes/SteamListingsOperation)

## Description

An overlapping SDK search returns busy.

## Example

```gdscript
extends Node

func run_fetch_operation(steam_lobbies: SteamLobbies, results_list: ItemList) -> void:
    var steam_lobby_query: SteamLobbyQuery = steam_lobbies.query()
    if steam_lobby_query == null:
        return
    var result: SteamListingsOperation = steam_lobby_query.fetch_operation()
    if result == null:
        return
    var result_value: SteamListingsResult = await result.wait()
    if result_value == null:
        return
    if not result_value.ok:
        push_warning(result_value.message)
        return
    var payload: Array[SteamListing] = result_value.value
    results_list.clear()
    for entry: SteamListing in payload:
        results_list.add_item(str(entry.name))
```
