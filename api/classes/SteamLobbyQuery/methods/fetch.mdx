## Example

```gdscript
const HARBOR_WORLD: StringName = &"harbor"
const REQUIRED_SEATS: int = 2

func list_harbor_lobbies(lobbies: SteamLobbies) -> void:
    var query: SteamLobbyQuery = lobbies.query().on_map(HARBOR_WORLD).with_room_for(REQUIRED_SEATS)
    var result: SteamListingsResult = await query.fetch()
    if not result.ok:
        push_warning(result.message)
        return
    print("Lobbies with two open seats: ", result.value.size())
```
