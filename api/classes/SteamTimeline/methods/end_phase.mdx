## Example

```gdscript
extends Node

func run_end_phase(steam_client: SteamClient, status_label: Label) -> void:
    if not steam_client.is_ready():
        return
    var steam_timeline: SteamTimeline = steam_client.timeline()
    if steam_timeline == null:
        return
    var result: Error = steam_timeline.end_phase()
    if result != OK:
        status_label.text = error_string(result)
        return
    status_label.text = "End phase accepted."
```
