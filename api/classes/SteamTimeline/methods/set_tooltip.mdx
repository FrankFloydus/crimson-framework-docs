## Example

```gdscript
extends Node

func run_set_tooltip(
    steam_client: SteamClient,
    description: String,
    offset_seconds: float,
    status_label: Label
) -> void:
    if not steam_client.is_ready():
        return
    var steam_timeline: SteamTimeline = steam_client.timeline()
    if steam_timeline == null:
        return
    var result: Error = steam_timeline.set_tooltip(description, offset_seconds)
    if result != OK:
        status_label.text = error_string(result)
        return
    status_label.text = "Set tooltip accepted."
```
