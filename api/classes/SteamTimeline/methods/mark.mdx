## Example

```gdscript
extends Node

func run_mark(
    steam_client: SteamClient,
    title: String,
    options: TimelineMarkerOptions,
    apply_mark: Callable
) -> void:
    if not steam_client.is_ready():
        return
    var steam_timeline: SteamTimeline = steam_client.timeline()
    if steam_timeline == null:
        return
    var result: SteamTimelineEventResult = steam_timeline.mark(title, options)
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: SteamTimelineEvent = result.value
    if payload == null:
        return
    apply_mark.call(payload)
```
