## Example

```gdscript
extends Node

func run_phase_recording(
    steam_client: SteamClient,
    phase_id: String,
    apply_phase_recording: Callable
) -> void:
    if not steam_client.is_ready():
        return
    var steam_timeline: SteamTimeline = steam_client.timeline()
    if steam_timeline == null:
        return
    var result: SteamTimelinePhaseResult = await steam_timeline.phase_recording(phase_id)
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: TimelinePhaseRecording = result.value
    if payload == null:
        return
    apply_phase_recording.call(payload)
```
