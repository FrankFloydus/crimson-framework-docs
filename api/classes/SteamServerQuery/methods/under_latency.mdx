## Example

```gdscript
extends Node

func run_under_latency(
    steam_server_browser: SteamServerBrowser,
    milliseconds: int,
    results_list: ItemList
) -> void:
    var steam_server_query: SteamServerQuery = steam_server_browser.query()
    if steam_server_query == null:
        return
    var result: SteamServerQuery = steam_server_query.under_latency(milliseconds)
    if result == null:
        return
    var completion: SteamListingsResult = await result.fetch()
    if completion == null:
        return
    if not completion.ok:
        push_warning(completion.message)
        return
    var payload: Array[SteamListing] = completion.value
    results_list.clear()
    for entry: SteamListing in payload:
        results_list.add_item(str(entry.name))
```
