## Example

```gdscript
extends Node

func run_is_available(steam_client: SteamClient, action_button: Button) -> void:
    if not steam_client.is_ready():
        return
    var steam_cloud: SteamCloud = steam_client.cloud()
    if steam_cloud == null:
        return
    var result: bool = steam_cloud.is_available()
    action_button.disabled = not result
```
