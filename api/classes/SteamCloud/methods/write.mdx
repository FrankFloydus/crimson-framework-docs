## Signature

```text
write(name: String, bytes: PackedByteArray) -> Variant
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `name` | `String` | — |
| `bytes` | `PackedByteArray` | — |

## Return value

`Variant`

Await the call to receive [`SteamResult`](/api/classes/SteamResult).

## Description

Use `write_file()` to retain an operation for cancellation.

## Example

### Resume a campaign at the last saved Steam Cloud checkpoint

Reaching a campaign checkpoint saves its ID to [Steam Cloud](/steam/client-services#save-a-profile-to-cloud). Press Continue to resume from the last saved checkpoint.

```gdscript
extends Control

signal continue_requested(checkpoint_id: StringName)

const CHECKPOINT_FILE: String = "campaign_checkpoint.json"
const CHECKPOINT_KEY: String = "checkpoint_id"

@export var steam_client: SteamClient
@export var continue_button: Button
@export var cloud_status: Label

var _cloud_busy: bool = false


func _ready() -> void:
    continue_button.pressed.connect(_continue_from_cloud)


func save_checkpoint(checkpoint_id: StringName) -> void:
    if not is_instance_valid(steam_client) or not steam_client.is_online():
        cloud_status.text = "Steam Cloud is unavailable."
        return
    if _cloud_busy:
        cloud_status.text = "Steam Cloud is busy. Try saving again."
        return

    _set_cloud_busy(true)
    var cloud: SteamCloud = steam_client.cloud()
    var payload: Dictionary = {CHECKPOINT_KEY: String(checkpoint_id)}
    var bytes: PackedByteArray = JSON.stringify(payload).to_utf8_buffer()
    var result: SteamResult = await cloud.write(CHECKPOINT_FILE, bytes)
    _set_cloud_busy(false)

    if result.ok:
        cloud_status.text = "Checkpoint saved to Steam Cloud."
    else:
        cloud_status.text = "Could not save checkpoint: %s" % result.message


func _continue_from_cloud() -> void:
    if not is_instance_valid(steam_client) or not steam_client.is_online():
        cloud_status.text = "Steam Cloud is unavailable."
        return

    _set_cloud_busy(true)
    var cloud: SteamCloud = steam_client.cloud()
    var result: SteamBytesResult = await cloud.read(CHECKPOINT_FILE)
    _set_cloud_busy(false)
    if not result.ok:
        cloud_status.text = "Could not load checkpoint: %s" % result.message
        return

    var decoded: Variant = JSON.parse_string(result.value.get_string_from_utf8())
    if not decoded is Dictionary:
        cloud_status.text = "The cloud checkpoint is invalid."
        return
    var payload: Dictionary = decoded
    var checkpoint_id: Variant = payload.get(CHECKPOINT_KEY)
    if not checkpoint_id is String or checkpoint_id.is_empty():
        cloud_status.text = "The cloud checkpoint has no checkpoint ID."
        return

    cloud_status.text = "Resuming from checkpoint %s." % checkpoint_id
    continue_requested.emit(StringName(checkpoint_id))


func _set_cloud_busy(is_busy: bool) -> void:
    _cloud_busy = is_busy
    continue_button.disabled = is_busy
```
