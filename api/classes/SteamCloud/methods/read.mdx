## Signature

```text
read(name: String) -> Variant
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `name` | `String` | — |

## Return value

`Variant`

Await the call to receive [`SteamBytesResult`](/api/classes/SteamBytesResult).

## Description

Inspect `SteamResult.ok` before reading `SteamBytesResult.value`. Use `read_file()` to retain an operation for cancellation.

## Example

```gdscript
extends Node

const PROFILE_FILE: String = "profile.json"
const DISPLAY_NAME_FIELD: String = "display_name"
const MAX_DISPLAY_NAME_LENGTH: int = 32

@export var steam_client: SteamClient
@export var profile_name: Label

# Connect this method to the Profile screen's Refresh button.
func refresh_profile_name() -> void:
    if steam_client == null or profile_name == null:
        return
    var cloud: SteamCloud = steam_client.cloud()
    if cloud == null:
        profile_name.text = "Steam Cloud unavailable"
        return
    var result: SteamBytesResult = await cloud.read(PROFILE_FILE)
    if not result.ok:
        profile_name.text = "Profile unavailable"
        return

    var parser: JSON = JSON.new()
    var parsed: Error = parser.parse(result.value.get_string_from_utf8())
    if parsed != OK or not parser.data is Dictionary:
        profile_name.text = "Invalid profile"
        return
    var profile: Dictionary = parser.data
    var display_name: Variant = profile.get(DISPLAY_NAME_FIELD)
    if not display_name is String:
        profile_name.text = "Player"
        return
    var cleaned_name: String = display_name.strip_edges().left(MAX_DISPLAY_NAME_LENGTH)
    profile_name.text = cleaned_name if not cleaned_name.is_empty() else "Player"
```
