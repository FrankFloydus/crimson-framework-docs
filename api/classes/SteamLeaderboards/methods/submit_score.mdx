## Example

```gdscript
extends Node

func run_submit_score(
    steam_client: SteamClient,
    board_name: String,
    score: int,
    apply_submit_score: Callable
) -> void:
    if not steam_client.is_ready():
        return
    var steam_leaderboards: SteamLeaderboards = steam_client.leaderboards()
    if steam_leaderboards == null:
        return
    var result: SteamLeaderboardSubmissionResult = await steam_leaderboards.submit_score(board_name, score)
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: SteamLeaderboardSubmission = result.value
    if payload == null:
        return
    apply_submit_score.call(payload)
```
