## Signature

```text
claim_promotions() -> Variant
```

## Return value

`Variant`

## Description

A successful result can contain no grants. Returns [`SteamInventoryChangeResult`](/api/classes/SteamInventoryChangeResult). Refresh inventory before retrying after a timeout.

## Example

```gdscript
extends Node

func run_claim_promotions(steam_client: SteamClient, apply_claim_promotions: Callable) -> void:
    if not steam_client.is_ready():
        return
    var steam_inventory: SteamInventory = steam_client.inventory()
    if steam_inventory == null:
        return
    var result: SteamInventoryChangeResult = await steam_inventory.claim_promotions()
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: SteamInventoryChange = result.value
    if payload == null:
        return
    apply_claim_promotions.call(payload)
```
