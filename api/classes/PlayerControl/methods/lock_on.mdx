## Return value

`Error`

## Description

The view keeps its own aim. The game chooses the target and handles any camera tracking. Returns `OK` on success, `ERR_UNCONFIGURED` if the controller is inactive, `ERR_UNAVAILABLE` for other styles, or `ERR_INVALID_PARAMETER` for a null, unmounted, queued-for-deletion, same-character, or different-viewport target. Falls back to view heading if the target leaves the viewport or has no horizontal separation. Network owners send steering heading only.

## Example

### Face a practice target and circle it

On an upright player with a camera and configured controls, choose **Lock-On** and assign the practice target. Press **L** to face it and circle, then press **L** again to return to moving relative to the view. [Set up the practice target](/player-controls#face-a-practice-target)

```gdscript
extends CharacterBody3D

const CAMERA_PATH: NodePath = ^"Camera3D"
const LOCK_KEY: Key = KEY_L

## The practice target this test can select.
@export var practice_target: Node3D

var _player_control: PlayerControl
var _target_locked: bool = false


func _ready() -> void:
    var controls: ControlsComponent = ControlsComponent.for_body(self)
    var camera: Camera3D = get_node_or_null(CAMERA_PATH) as Camera3D
    if controls == null or camera == null:
        push_error("Add one ControlsComponent and a Camera3D named Camera3D.")
        return

    _player_control = controls.boot(camera)
    if _player_control == null:
        push_error("The player controls could not attach.")
    elif not _player_control.is_ready():
        push_error(_player_control.get_message())


func _unhandled_key_input(event: InputEvent) -> void:
    var key: InputEventKey = event as InputEventKey
    if key == null or not key.pressed or key.echo or key.keycode != LOCK_KEY:
        return
    if _player_control == null or not _player_control.is_ready():
        return

    if _target_locked:
        _player_control.clear_lock_on()
        _target_locked = false
        return

    var result: Error = _player_control.lock_on(practice_target)
    if result != OK:
        push_warning("Could not face the practice target: %s" % error_string(result))
        return

    _target_locked = true
```
