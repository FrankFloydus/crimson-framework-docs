## Signature

```text
add_change(store_id: String, operation: Dictionary, expected_revision: int) -> Error
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `store_id` | `String` | — |
| `operation` | `Dictionary` | — |
| `expected_revision` | `int` | — |

## Return value

`Error`

## Description

The coordinator checks that revision during commit. Returns `ERR\_UNAVAILABLE` for an unowned, consumed, restore-only, or wrong-thread handle, and `ERR\_INVALID\_DATA` if the operation exceeds the data limits or contains unsupported values.

## Example

```gdscript
extends Node

const PICKUP_STORE: String = "world_pickups"
const WALLET_STORE: String = "player_wallet"
const COIN_POUCH: String = "coin_pouch_at_gate"
const POUCH_COINS: int = 8

func _ready() -> void:
    var coordinator: TransactionCoordinator = TransactionCoordinator.new()
    var registered: Error = coordinator.register_store(
        PICKUP_STORE, {COIN_POUCH: true}, _prepare_pickups
    )
    if registered == OK:
        registered = coordinator.register_store(WALLET_STORE, {"coins": 0}, _prepare_wallet)
    if registered != OK:
        push_error(error_string(registered))
        return

    # Run these changes together when the authority awards a pickup.
    var collect: GameTransaction = coordinator.begin()
    var pickup_revision: int = coordinator.get_store_revision(PICKUP_STORE)
    var wallet_revision: int = coordinator.get_store_revision(WALLET_STORE)
    var queued: Error = collect.add_change(PICKUP_STORE, {"take": COIN_POUCH}, pickup_revision)
    if queued == OK:
        queued = collect.add_change(WALLET_STORE, {"add": POUCH_COINS}, wallet_revision)
    if queued != OK:
        push_warning(error_string(queued))
        return

    var committed: Dictionary = coordinator.commit(collect)
    if not committed.get("ok", false):
        push_warning(String(committed.get("message", "Pickup rejected.")))
        return

    var pickups: Dictionary = coordinator.read_store(PICKUP_STORE)
    var wallet: Dictionary = coordinator.read_store(WALLET_STORE)
    print("Pickup still visible: ", pickups[COIN_POUCH], "; HUD coins: ", wallet["coins"])

func _prepare_pickups(state: Dictionary, operations: Array, _context: Dictionary) -> Dictionary:
    for operation: Dictionary in operations:
        var pickup_id: String = String(operation["take"])
        if not state.get(pickup_id, false):
            return {"ok": false, "message": "This pickup was already collected."}
        state[pickup_id] = false
    return {"ok": true, "state": state}

func _prepare_wallet(state: Dictionary, operations: Array, _context: Dictionary) -> Dictionary:
    for operation: Dictionary in operations:
        var amount: int = int(operation["add"])
        if amount <= 0:
            return {"ok": false, "message": "A pickup must award a positive amount."}
        state["coins"] = int(state["coins"]) + amount
    return {"ok": true, "state": state}
```
