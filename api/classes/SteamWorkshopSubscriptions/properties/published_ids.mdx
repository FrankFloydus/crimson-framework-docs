## Example

### Show subscribed Workshop maps in the custom map browser

Press Refresh in Custom Maps to load titles for maps the player subscribes to in Steam Workshop. Selecting a map passes its published ID to the game.

```gdscript
extends Control

signal map_selected(published_id: String)

const MAPS_PER_REFRESH: int = 12

@export var steam_client: SteamClient
@export var refresh_button: Button
@export var map_list: ItemList
@export var workshop_status: Label


func _ready() -> void:
    refresh_button.pressed.connect(_refresh_maps)
    map_list.item_selected.connect(_select_map)


func _refresh_maps() -> void:
    map_list.clear()
    if not is_instance_valid(steam_client) or not steam_client.is_online():
        workshop_status.text = "Sign in to Steam to see your maps."
        return

    var workshop: SteamWorkshop = steam_client.workshop()
    var result: SteamWorkshopSubscriptionsResult = workshop.subscriptions(MAPS_PER_REFRESH)
    if not result.ok:
        workshop_status.text = "Could not load subscriptions: %s" % result.message
        return

    var subscriptions: SteamWorkshopSubscriptions = result.value
    if subscriptions.published_ids.is_empty():
        workshop_status.text = "Subscribe to a map in the Steam Workshop to see it here."
        return

    refresh_button.disabled = true
    workshop_status.text = "Loading map titles..."
    for published_id: String in subscriptions.published_ids:
        var item: SteamWorkshopItemRef = workshop.item(published_id)
        var details: SteamWorkshopItemResult = await item.fetch()
        if not details.ok:
            continue
        var row: int = map_list.add_item(details.value.title)
        map_list.set_item_metadata(row, published_id)
    refresh_button.disabled = false
    workshop_status.text = "Showing %d of %d subscribed maps." % [
        map_list.item_count, subscriptions.total_count,
    ]


func _select_map(index: int) -> void:
    var published_id: String = map_list.get_item_metadata(index)
    map_selected.emit(published_id)
```
