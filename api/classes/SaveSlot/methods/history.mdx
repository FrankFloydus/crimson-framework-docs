## Signature

```text
history(before_sequence: String = "", before_id: String = "", limit: int = 16) -> GameOperation
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `before_sequence` | `String` | `""` |
| `before_id` | `String` | `""` |
| `limit` | `int` | `16` |

## Return value

[`GameOperation`](/api/classes/GameOperation)

## Description

Leave both cursor values empty for the first page or pass both values from the previous result's next cursor. The page limit is positive and capped at 16. On success, the operation reports the rows in value and the slot status.

## Example

```gdscript
extends Node

const PROFILE_ID: String = "local_player"
const SLOT_ID: String = "campaign"
const PAGE_SIZE: int = 8

# Assign an attached, configured GameRuntime and the load menu's ItemList.
@export var game: GameRuntime
@export var checkpoint_list: ItemList

func _ready() -> void:
    if game == null or checkpoint_list == null:
        push_error("Assign the game runtime and checkpoint list.")
        return

    var slot: SaveSlot = game.save_slot(PROFILE_ID, SLOT_ID)
    if slot == null:
        push_warning("This game has no checkpoint slot.")
        return
    var operation: GameOperation = slot.history("", "", PAGE_SIZE)
    var page: Dictionary = await operation.wait()
    if not page.get("ok", false):
        push_warning(String(page.get("message", "Could not read checkpoint history.")))
        return

    checkpoint_list.clear()
    var generations: Array = page.get("value", [])
    for generation: Dictionary in generations:
        var display: Dictionary = SavePresentation.describe(generation)
        var label: String = String(display.get("label", ""))
        if label.is_empty():
            label = "Checkpoint " + String(generation["sequence"])
        var row: int = checkpoint_list.add_item(label)
        # Keep both values so the menu can restore this exact generation.
        checkpoint_list.set_item_metadata(row, {
            "sequence": generation["sequence"],
            "id": generation["id"],
        })
```
