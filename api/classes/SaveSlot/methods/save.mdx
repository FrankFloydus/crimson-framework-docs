## Signature

```text
save() -> SaveRequest
```

## Return value

[`SaveRequest`](/api/classes/SaveRequest)

## Description

Call submit on the request to queue the save.

## Example

```gdscript
extends Node

const PROFILE: String = "local-player"
const SLOT: String = "campaign"
const CHECKPOINT_NAME: String = "Harbor checkpoint"
const LOCATION: String = "Harbor"

@export var game: GameRuntime
@export var save_status: Label

# Connect the Save button when the coordinated authority is running.
func save_harbor_checkpoint() -> void:
    if game == null or save_status == null:
        return
    var slot: SaveSlot = game.save_slot(PROFILE, SLOT)
    var request: SaveRequest = slot.save()
    request = request.named(CHECKPOINT_NAME).with_context({"location": LOCATION})
    save_status.text = "Saving checkpoint…"
    var operation: GameOperation = request.submit()
    var saved: Dictionary = await operation.wait()
    if not saved.ok:
        save_status.text = saved.message
        return
    save_status.text = "Saved " + CHECKPOINT_NAME
```
