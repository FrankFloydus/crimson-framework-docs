## Example

### Publish a dedicated arena and report its readiness

Start your [dedicated Steam arena](/steam/dedicated-servers#start-a-playerless-authority) to print its join locator. The server logs changes in readiness, secure status, advertising, and the verified player count.

```gdscript
extends Node

const APP_ID: int = 480
const GAME_PORT: int = 27015
const QUERY_PORT: int = 27016
const NAMESPACE: String = "my-studio-arena"
const SERVER_NAME: String = "Cargo Yard Arena"
const PRODUCT_VERSION: String = "1.0.0.0"
const GAME_DESCRIPTION: String = "Spacewar"
const INSTALL_DIR: String = "Spacewar"

@export var game_definition: GameDefinition

var _server: SteamDedicatedServer


func _ready() -> void:
    var settings: SteamServer = SteamServer.configure(SERVER_NAME, GAME_PORT, QUERY_PORT)
    settings = settings.with_product_metadata(PRODUCT_VERSION, GAME_DESCRIPTION, INSTALL_DIR)
    settings = settings.advertise_on(SteamServer.INTERNET)

    var provider: SteamProvider = SteamProvider.configure().for_app(APP_ID)
    provider = provider.with_lobby_namespace(NAMESPACE).dedicated(settings)
    var game: GameRuntime = game_definition.with_provider(provider).boot(self)
    if game == null:
        push_error("Could not create the arena server.")
        return

    var started: Dictionary = await game.start_dedicated_online().wait()
    if not started.ok:
        push_error("Could not open the arena. %s" % started.message)
        return
    print("Arena server locator: ", game.get_online_listing())

    _server = SteamServices.server_for(game)
    if _server == null:
        push_error("The arena has no Steam server attachment.")
        return
    _server.status_changed.connect(_report_status)
    _report_status(_server.status())


func _report_status(status: SteamServerStatus) -> void:
    print("Arena %s | Ready: %s | Steam secure: %s | Advertising requested: %s | Players: %d" % [
        status.world_id, status.ready, status.secure,
        status.advertisement_requested, status.authenticated_players
    ])
```
