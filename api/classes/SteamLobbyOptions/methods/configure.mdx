## Example

```gdscript
extends Node

func run_configure(
    steam_client: SteamClient,
    create_lobby_confirmed: bool,
    apply_create: Callable
) -> void:
    var result: SteamLobbyOptions = SteamLobbyOptions.configure()
    if result == null or not steam_client.is_ready() or not create_lobby_confirmed:
        return
    var lobbies: SteamLobbies = steam_client.lobbies()
    if lobbies == null:
        return
    var operation: SteamLobbyOperation = lobbies.create(result)
    if operation == null:
        return
    var result_value: SteamLobbyResult = await operation.wait()
    if result_value == null:
        return
    if not result_value.ok:
        push_warning(result_value.message)
        return
    var payload: SteamLobby = result_value.value
    if payload == null:
        return
    apply_create.call(payload)
```
