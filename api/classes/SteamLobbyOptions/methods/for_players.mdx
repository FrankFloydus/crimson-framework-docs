## Example

### Open a Steam practice room and show who joins

Press Create room in the Practice menu, then invite friends through the Steam Overlay. The [social lobby](/steam/listen-servers#run-a-social-lobby-without-a-game) updates the roster as players join or leave.

```gdscript
extends Control

const PRACTICE_NAMESPACE: StringName = &"arena-practice"
const ROOM_TITLE: String = "Arena practice room"
const ROOM_SIZE: int = 4

@export var steam_client: SteamClient
@export var create_button: Button
@export var invite_button: Button
@export var leave_button: Button
@export var member_list: ItemList
@export var room_status: Label

var _lobby: SteamLobby


func _ready() -> void:
    _set_room_controls(false)
    create_button.pressed.connect(_create_room)
    invite_button.pressed.connect(_invite_friends)
    leave_button.pressed.connect(_leave_room)


func _create_room() -> void:
    if not is_instance_valid(steam_client) or not steam_client.is_online():
        room_status.text = "Sign in to Steam to open a practice room."
        return

    create_button.disabled = true
    room_status.text = "Opening practice room..."
    var lobbies: SteamLobbies = steam_client.lobbies()
    var options: SteamLobbyOptions = (
        SteamLobbyOptions.configure()
        .in_namespace(PRACTICE_NAMESPACE)
        .for_players(ROOM_SIZE)
        .with_visibility(SteamLobbyOptions.PRIVATE)
    )
    var result: SteamLobbyResult = await lobbies.create(options).wait()
    if not result.ok:
        room_status.text = "Could not open practice room: %s" % result.message
        _set_room_controls(false)
        return

    _lobby = result.value
    _lobby.closed.connect(_on_room_closed)
    _lobby.members_changed.connect(_show_members)
    _set_room_controls(true)
    _show_members(_lobby.members())

    var edit: SteamLobbyEdit = _lobby.edit().named(ROOM_TITLE)
    var rename_error: Error = edit.submit()
    room_status.text = ROOM_TITLE if rename_error == OK else "Practice room opened without a title."


func _show_members(members: Array[SteamLobbyMember]) -> void:
    member_list.clear()
    for member: SteamLobbyMember in members:
        var player_name: String = member.display_name
        if player_name.is_empty():
            player_name = "Steam player %s" % member.steam_id
        if member.is_owner:
            player_name += " (host)"
        member_list.add_item(player_name)


func _invite_friends() -> void:
    var result: Error = _lobby.open_invite_dialog()
    if result == OK:
        room_status.text = "Choose a friend in the Steam Overlay."
    else:
        room_status.text = "Could not open the invite picker: %s" % error_string(result)


func _leave_room() -> void:
    var result: Error = _lobby.leave()
    if result != OK:
        room_status.text = "Could not leave the room: %s" % error_string(result)


func _on_room_closed(reason: int) -> void:
    _lobby = null
    member_list.clear()
    _set_room_controls(false)
    room_status.text = "You left the practice room." if reason == OK else "Practice room closed."


func _set_room_controls(in_room: bool) -> void:
    create_button.disabled = in_room
    invite_button.disabled = not in_room
    leave_button.disabled = not in_room
```
