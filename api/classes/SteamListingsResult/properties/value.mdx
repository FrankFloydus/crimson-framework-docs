## Example

```gdscript
extends Node

func refresh_value(steam_listings_result: SteamListingsResult, results_list: ItemList) -> void:
    if steam_listings_result == null:
        return
    if not steam_listings_result.ok:
        push_warning(steam_listings_result.message)
        return
    var value: Array[SteamListing] = steam_listings_result.value
    results_list.clear()
    for entry: SteamListing in value:
        results_list.add_item(str(entry.name))
```
