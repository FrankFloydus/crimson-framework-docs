## Example

```gdscript
extends SteamPublisherHost

const PUBLISHER_KEY_REFERENCE: StringName = &"STEAM_PUBLISHER_KEY"

func _resolve_secret(reference: StringName) -> String:
    if reference != PUBLISHER_KEY_REFERENCE:
        return ""
    return OS.get_environment(String(PUBLISHER_KEY_REFERENCE))
```
