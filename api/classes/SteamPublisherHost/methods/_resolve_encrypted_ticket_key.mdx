## Description

Resolve this key only inside the trusted backend.

## Example

```gdscript
extends SteamPublisherHost

const ENCRYPTED_TICKET_KEY_BYTES: int = 32

# Set this to a backend-only secret-store lookup that returns raw key bytes.
var load_ticket_key: Callable

func _resolve_encrypted_ticket_key(reference: StringName) -> PackedByteArray:
    if not load_ticket_key.is_valid():
        return PackedByteArray()
    var key: PackedByteArray = load_ticket_key.call(reference)
    return key if key.size() == ENCRYPTED_TICKET_KEY_BYTES else PackedByteArray()
```
