## Return value

[`NetworkEncodeResult`](/api/classes/NetworkEncodeResult)

## Description

The result contains the complete payload only when the callback returns `OK` and all writes succeed. An unknown registration returns `ERR_DOES_NOT_EXIST`. A callback invocation error or invalid return type returns `ERR_INVALID_DECLARATION`.

## Example

### Encode an ammo update after a rifle burst

Create a level and player scene, then run the script on a plain Node. It logs the remaining rounds and the four-byte payload after the update round-trips. [Ammo update setup](/binary-messages#encode-an-ammo-update)

```gdscript
extends Node

const LEVEL: PackedScene = preload("res://world.tscn")
const PLAYER: PackedScene = preload("res://player.tscn")
const AMMO_CODEC: StringName = &"weapons.ammo"
const AMMO_VERSION: int = 1
const AMMO_BYTES: int = 4
const RIFLE_ID: int = 7
const RIFLE_CAPACITY: int = 30
const ROUNDS_AFTER_BURST: int = 18

class AmmoUpdate extends RefCounted:
    var weapon_id: int
    var rounds: int
    var reloading: bool

var _game: Game


func _ready() -> void:
    _game = Game.configure(LEVEL, PLAYER).boot(self)
    if not _game.boot_result().ok:
        push_error(_game.boot_result().message)
        return

    var codecs: NetworkCodecs = _game.codecs()
    var registered: Error = codecs.register_codec(AMMO_CODEC, AMMO_VERSION, _write_ammo, _read_ammo)
    if registered != OK:
        push_error("Could not register the ammo codec. %s" % error_string(registered))
        return

    var update: AmmoUpdate = AmmoUpdate.new()
    update.weapon_id = RIFLE_ID
    update.rounds = ROUNDS_AFTER_BURST
    var encoded: NetworkEncodeResult = codecs.encode(AMMO_CODEC, AMMO_VERSION, update, AMMO_BYTES)
    if not encoded.ok:
        push_error("Could not encode the ammo update. %s" % error_string(encoded.error))
        return

    var decoded: NetworkDecodeResult = codecs.decode(AMMO_CODEC, AMMO_VERSION, encoded.bytes, AMMO_BYTES)
    if not decoded.ok:
        push_error("Could not decode the ammo update. %s" % error_string(decoded.error))
        return

    var received: AmmoUpdate = decoded.value as AmmoUpdate
    if received == null:
        push_error("The ammo codec returned an unexpected value.")
        return
    print("Rifle magazine: %d rounds (%d bytes)." % [received.rounds, encoded.bytes.size()])


func _write_ammo(writer: NetworkWriter, value: Variant) -> Error:
    var update: AmmoUpdate = value as AmmoUpdate
    if update == null or update.weapon_id != RIFLE_ID or update.rounds < 0 or update.rounds > RIFLE_CAPACITY:
        return ERR_INVALID_DATA
    writer.write_u16(update.weapon_id)
    writer.write_u8(update.rounds)
    return writer.write_bool(update.reloading)


func _read_ammo(reader: NetworkReader) -> Variant:
    var update: AmmoUpdate = AmmoUpdate.new()
    update.weapon_id = reader.read_u16()
    update.rounds = reader.read_u8()
    update.reloading = reader.read_bool()
    if update.weapon_id != RIFLE_ID or update.rounds > RIFLE_CAPACITY:
        reader.fail()
        return null
    return update
```
