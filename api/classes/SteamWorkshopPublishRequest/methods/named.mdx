```gdscript
extends Node

func run_named(
    steam_workshop_publish_request: SteamWorkshopPublishRequest,
    title: String,
    confirmed: bool,
    published_id_label: Label
) -> void:
    var result: SteamWorkshopPublishRequest = steam_workshop_publish_request.named(title)
    if result == null:
        return
    if not confirmed:
        return
    var completion: SteamWorkshopPublicationResult = await result.submit()
    if completion == null:
        return
    if not completion.ok:
        push_warning(completion.message)
        return
    var payload: SteamWorkshopPublication = completion.value
    if payload == null:
        return
    var published_id: String = payload.published_id
    published_id_label.text = "Published id: %s" % str(published_id)
```
