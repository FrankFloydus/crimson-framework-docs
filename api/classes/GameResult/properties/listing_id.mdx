## Example

### Host an invite-only Steam arena

Press Host to open a two-player arena, then Invite to choose a Steam friend. Launch the game through Steam.

```gdscript
extends Node

const TOTAL_SEATS: int = 2

@export var arena: PackedScene
@export var player: PackedScene
@export var host_button: Button
@export var invite_button: Button
@export var status: Label

var _game: Game

func _ready() -> void:
    host_button.disabled = true
    invite_button.disabled = true
    var setup: GameConfig = Game.configure(arena, player).steam(null, TOTAL_SEATS)
    _game = setup.boot(self)
    var booted: GameResult = _game.boot_result()
    if not booted.ok:
        status.text = booted.message
        return
    var steam: Steamworks = _game.steam()
    if not steam.is_ready():
        status.text = steam.boot_result().message
        return
    host_button.pressed.connect(_host)
    invite_button.pressed.connect(_invite)
    host_button.disabled = false

func _host() -> void:
    host_button.disabled = true
    var hosted: GameResult = await _game.host()
    if not hosted.ok:
        status.text = hosted.message
        host_button.disabled = false
        return
    status.text = "Arena open. Lobby %s." % hosted.listing_id
    invite_button.disabled = false

func _invite() -> void:
    var requested: GameResult = _game.invite_friends()
    status.text = "Choose a friend in Steam." if requested.ok else requested.message
```
