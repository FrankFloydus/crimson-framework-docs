## Signature

```text
edit() -> SteamStatsEdit
```

## Return value

[`SteamStatsEdit`](/api/classes/SteamStatsEdit)

## Description

This call has no effect on the Steam cache.

## Example

```gdscript
extends Node

# Define this integer stat in Steamworks for your app.
const MATCHES_STAT: String = "matches_played"

@export var steam_client: SteamClient
@export var stats_status: Label

# Call after a match. Pass the player's new absolute match count.
func record_matches_played(total_matches: int) -> void:
    if steam_client == null or stats_status == null:
        return
    var stats: SteamStats = steam_client.stats()
    if stats == null:
        stats_status.text = "Steam stats unavailable"
        return

    var draft: SteamStatsEdit = stats.edit()
    draft = draft.set_int(MATCHES_STAT, total_matches)
    var submitted: SteamStatsSubmissionResult = draft.submit()
    if not submitted.ok:
        stats_status.text = submitted.message
        return
    # Observe stats_stored before reporting that Steam stored the change.
    stats_status.text = "Stat changes accepted locally: %d" % submitted.value.applied
```
