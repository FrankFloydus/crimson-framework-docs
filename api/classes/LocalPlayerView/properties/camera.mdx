## Description

Pick a Camera3D from this same player scene.

## Example

### Choose first-person or chase view before spawning

Choose first-person or chase view before entering a solo range. The first-person view hides the player's body while the chase view keeps it visible.

```gdscript
extends Node

const TRAINING_ROOM: StringName = &"training_room"
const FIRST_PERSON_CAMERA: NodePath = ^"Eyes"
const CHASE_CAMERA: NodePath = ^"ChaseCamera"
const BODY: NodePath = ^"Body"
const LOCAL_VIEW: NodePath = ^"LocalPlayerView"

@export var room_scene: PackedScene
@export var player_template: PackedScene
@export var menu: Control
@export var first_person_mode: CheckButton
@export var play_button: Button
@export var status: Label


func _ready() -> void:
    play_button.pressed.connect(_play)


func _play() -> void:
    if room_scene == null or player_template == null:
        status.text = "Assign the training room and the player template."
        return

    var player_scene: PackedScene = _make_player_scene(first_person_mode.button_pressed)
    if player_scene == null:
        return

    play_button.disabled = true
    var definition: GameDefinition = (
        GameRuntime.configure()
        .with_starting_world(TRAINING_ROOM, room_scene)
        .with_player(player_scene)
    )
    var game: GameRuntime = definition.boot(self)
    if game == null:
        status.text = "Could not create the game."
        play_button.disabled = false
        return

    var result: Dictionary = await game.play_solo().wait()
    if not result.ok:
        status.text = "Could not enter the training room: %s" % result.message
        game.queue_free()
        await game.tree_exited
        play_button.disabled = false
        return

    menu.hide()


func _make_player_scene(first_person: bool) -> PackedScene:
    var actor: Node = player_template.instantiate()
    if actor == null:
        status.text = "Could not instantiate the player template."
        return null

    var camera_path: NodePath = FIRST_PERSON_CAMERA if first_person else CHASE_CAMERA
    var camera: Camera3D = actor.get_node_or_null(camera_path) as Camera3D
    var body: Node3D = actor.get_node_or_null(BODY) as Node3D
    var view: LocalPlayerView = actor.get_node_or_null(LOCAL_VIEW) as LocalPlayerView
    if actor is not Node3D or camera == null or body == null or view == null:
        status.text = "The player template needs both cameras, a body, and a LocalPlayerView."
        actor.free()
        return null

    view.camera = camera
    view.hidden_visuals = body if first_person else null

    # Save the view references in the scene that Crimson will spawn.
    var player_scene: PackedScene = PackedScene.new()
    var error: Error = player_scene.pack(actor)
    actor.free()
    if error != OK:
        status.text = "Could not prepare the player scene: %s" % error_string(error)
        return null
    return player_scene
```
