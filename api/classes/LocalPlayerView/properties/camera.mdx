## Description

Pick a Camera3D from this same player scene.

## Example

```gdscript
extends Node

const WORLD_ID: StringName = &"training_room"
const CAMERA_HEIGHT: float = 1.6
const VIEW_PATH: NodePath = ^"LocalPlayerView"

# Assign a Node3D world scene with at least one PlayerSpawnPoint.
@export var world_scene: PackedScene

func _ready() -> void:
    if world_scene == null:
        push_error("Assign the training-room scene.")
        return

    var player_scene: PackedScene = _make_player_scene()
    if player_scene == null:
        return

    var definition: GameDefinition = GameRuntime.configure()
    definition = definition.with_starting_world(WORLD_ID, world_scene)
    definition = definition.with_player(player_scene)
    var game: GameRuntime = definition.boot(self)
    if game == null:
        push_error("Could not create the game.")
        return
    var operation: GameOperation = game.play_solo()
    var started: Dictionary = await operation.wait()
    if not started.get("ok", false):
        push_warning(String(started.get("message", "Could not start the game.")))
        return

    var local_player: Node3D = game.get_local_player()
    if local_player == null:
        return
    var local_view: LocalPlayerView = local_player.get_node(VIEW_PATH)
    var active_camera: Camera3D = local_view.camera
    if active_camera != null and active_camera.is_current():
        print("The local player is looking through ", active_camera.name)

func _make_player_scene() -> PackedScene:
    var actor: Node3D = Node3D.new()
    var eyes: Camera3D = Camera3D.new()
    eyes.name = &"FirstPersonCamera"
    eyes.position.y = CAMERA_HEIGHT
    actor.add_child(eyes)
    eyes.owner = actor
    # Pack the view with the player so its references exist before spawning.
    var view: LocalPlayerView = LocalPlayerView.new()
    view.name = &"LocalPlayerView"
    actor.add_child(view)
    view.owner = actor
    view.camera = eyes
    var packed: PackedScene = PackedScene.new()
    var packed_status: Error = packed.pack(actor)
    actor.free()
    if packed_status != OK:
        push_error(error_string(packed_status))
        return null
    return packed
```
