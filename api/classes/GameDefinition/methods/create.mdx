## Signature

```text
create() -> GameRuntime
```

## Return value

[`GameRuntime`](/api/classes/GameRuntime)

## Description

Add it to a persistent scene before starting a session.

## Example

```gdscript
extends Node

@export var game_definition: GameDefinition
@export var game_container: Node

# Use this when the game must live under a specific scene node.
func _ready() -> void:
    if game_definition == null or game_container == null:
        push_error("Assign the game definition and its container.")
        return
    var game: GameRuntime = game_definition.create()
    game_container.add_child(game)
    var started: Dictionary = await game.play_solo().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("Game mounted under ", game_container.name)
```
