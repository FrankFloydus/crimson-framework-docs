## Return value

[`GameDefinition`](/api/classes/GameDefinition)

## Description

For coordinated play, use an ID from the world catalog.

## Example

```gdscript
extends Node

const FOUNDRY_WORLD: StringName = &"foundry"

# Assign a definition whose Worlds catalog includes the foundry.
@export var campaign: GameDefinition

func _ready() -> void:
    if campaign == null:
        push_error("Assign the campaign definition.")
        return
    var foundry_start: GameDefinition = campaign.starting_in(FOUNDRY_WORLD)
    var game: GameRuntime = foundry_start.boot(self)
    if game == null:
        return
    var started: Dictionary = await game.start_offline().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("Campaign starts in ", game.get_world_id())
```
