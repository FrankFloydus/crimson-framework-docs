
## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `worlds` | `Dictionary` | — |

## Return value

[`GameDefinition`](/api/classes/GameDefinition)

## Description

Map each stable world ID to a dictionary containing scene\_id and a local res:// path to a .tscn or .scn file.

## Example

```gdscript
extends Node

const HARBOR_WORLD: StringName = &"harbor"
const FOUNDRY_WORLD: StringName = &"foundry"
const HARBOR_SCENE_ID: StringName = &"harbor_map"
const FOUNDRY_SCENE_ID: StringName = &"foundry_map"

@export_file("*.tscn") var harbor_path: String
@export_file("*.tscn") var foundry_path: String

# Assign the two scene files in the Inspector.
func _ready() -> void:
    if harbor_path.is_empty() or foundry_path.is_empty():
        push_error("Choose the harbor and foundry scenes.")
        return
    var worlds: Dictionary = {
        HARBOR_WORLD: {"scene_id": HARBOR_SCENE_ID, "path": harbor_path},
        FOUNDRY_WORLD: {"scene_id": FOUNDRY_SCENE_ID, "path": foundry_path},
    }
    var definition: GameDefinition = GameDefinition.configure()
    definition = definition.with_worlds(worlds).starting_in(HARBOR_WORLD)
    var game: GameRuntime = definition.boot(self)
    if game == null:
        return
    var started: Dictionary = await game.start_offline().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("The campaign is ready in ", game.get_world_id())
```
