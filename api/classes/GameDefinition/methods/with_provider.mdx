## Signature

```text
with_provider(provider: GameServiceProvider) -> GameDefinition
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `provider` | [`GameServiceProvider`](/api/classes/GameServiceProvider) | — |

## Return value

[`GameDefinition`](/api/classes/GameDefinition)

## Description

Pass null to use the default session and scene loader.

## Example

```gdscript
extends Node

const STEAM_APP_ID: int = 480
const LOBBY_NAMESPACE: String = "crimson-harbor"
const TOTAL_SEATS: int = 2

# The world needs at least two player spawn points for this capacity.
@export var game_definition: GameDefinition

# Connect a Host button after setting the world and player scenes in the definition.
func host_friends_game() -> void:
    if game_definition == null:
        push_error("Assign a game definition.")
        return
    var steam: SteamProvider = SteamProvider.configure().for_app(STEAM_APP_ID)
    steam = steam.with_lobby_namespace(LOBBY_NAMESPACE).friends_only()
    var rules: MultiplayerRules = MultiplayerRules.listen_coop().with_max_players(TOTAL_SEATS)
    var definition: GameDefinition = game_definition.with_provider(steam).with_multiplayer(rules)
    var game: GameRuntime = definition.boot(self)
    if game == null:
        return
    var hosted: Dictionary = await game.host_online().wait()
    if not hosted.ok:
        push_error(hosted.message)
        game.queue_free()
        return
    print("Steam friends lobby ready: ", game.get_online_listing())
```
