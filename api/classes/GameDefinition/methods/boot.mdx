## Return value

[`GameRuntime`](/api/classes/GameRuntime)

## Description

Returns null if attachment fails or the tree retires the owner or child during attachment.

## Example

```gdscript
extends Node

@export var game_definition: GameDefinition
@export var play_button: Button

var _game: GameRuntime

# Assign both exports. The button is enabled only after the game attaches.
func _ready() -> void:
    if game_definition == null or play_button == null:
        push_error("Assign the game definition and the Play button.")
        return
    play_button.disabled = true
    _game = game_definition.boot(self)
    if _game == null:
        push_error("Could not attach the game.")
        return
    play_button.pressed.connect(_enter_game)
    play_button.disabled = false

func _enter_game() -> void:
    play_button.disabled = true
    var started: Dictionary = await _game.play_solo().wait()
    if not started.ok:
        push_error(started.message)
        play_button.disabled = false
```
