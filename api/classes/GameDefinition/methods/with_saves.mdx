## Description

Choose a stable project-specific app\_id and a writable path under user://.

## Example

```gdscript
extends Node

const SAVE_APP_ID: String = "harbor.example.campaign"
const SAVE_DIRECTORY: String = "user://harbor-checkpoints"
const SAVE_SCHEMA: int = 1
const PROFILE: String = "local-player"
const SLOT: String = "campaign"

# Assign a coordinated campaign definition with a registered starting world.
@export var campaign: GameDefinition
var _game: GameRuntime

func _ready() -> void:
    if campaign == null:
        push_error("Assign the campaign definition.")
        return
    var definition: GameDefinition = campaign.with_saves(SAVE_APP_ID, SAVE_DIRECTORY, SAVE_SCHEMA)
    _game = definition.boot(self)
    if _game == null:
        return
    var started: Dictionary = await _game.start_offline().wait()
    if not started.ok:
        push_error(started.message)

# Connect a Save button after the game reports that its world is ready.
func save_checkpoint() -> void:
    if not is_instance_valid(_game):
        return
    var saved: Dictionary = await _game.save(PROFILE, SLOT).wait()
    if not saved.ok:
        push_error(saved.message)
        return
    print("Saved to the campaign slot.")
```
