## Description

Place PlayerSpawnPoint nodes in this scene to supply player placements.

## Example

```gdscript
extends Node

const HARBOR_WORLD: StringName = &"harbor"

@export var harbor: PackedScene
@export var player: PackedScene

func _ready() -> void:
    if harbor == null or player == null:
        push_error("Assign the harbor and player scenes.")
        return
    var definition: GameDefinition = GameDefinition.configure()
    definition.starting_world = HARBOR_WORLD
    # The harbor scene needs a Node3D root and an authored spawn placement.
    definition.starting_world_scene = harbor
    definition.player_scene = player
    var game: GameRuntime = definition.boot(self)
    if game == null:
        return
    var started: Dictionary = await game.play_solo().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("Mounted scene: ", game.get_world().name)
```
