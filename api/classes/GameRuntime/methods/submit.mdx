## Example

```gdscript
const USE_DOOR_COMMAND: String = "use_door"

func use_door(game: GameRuntime, door_id: StringName) -> void:
    # The authority must register this command before clients submit it.
    var result: Dictionary = await game.submit(USE_DOOR_COMMAND, {"door_id": door_id}).wait()
    if not result.ok:
        push_warning(result.message)
```
