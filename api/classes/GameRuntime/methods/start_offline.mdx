## Example

```gdscript
extends Node

func run_start_offline(game_runtime: GameRuntime, status_label: Label) -> void:
    var result: GameOperation = game_runtime.start_offline()
    if result == null:
        return
    var result_value: Dictionary = await result.wait()
    if not result_value.get("ok", false):
        status_label.text = str(result_value.get("message", "Operation failed."))
        return
    status_label.text = "Start offline completed."
```
