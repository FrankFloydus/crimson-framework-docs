## Signature

```text
start_dedicated(bind_address: String = "*") -> GameOperation
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `bind_address` | `String` | `"*"` |

## Return value

[`GameOperation`](/api/classes/GameOperation)

## Description

All configured player seats are available to guests. Await the operation for world and admission readiness.

## Example

```gdscript
extends Node

const EXIT_FAILURE: int = 1
const TEST_BIND_ADDRESS: String = "127.0.0.1"

# Attach to a headless Cargo Yard server scene. Assign a definition with
# the yard, its player scene, and one spawn point per multiplayer seat.
# Enable join in progress so players can enter the running server.
@export var game_definition: GameDefinition

func _ready() -> void:
    if game_definition == null:
        push_error("Assign the cargo yard server definition.")
        get_tree().quit(EXIT_FAILURE)
        return
    var game: GameRuntime = game_definition.boot(self)
    if game == null:
        push_error("Could not attach the cargo yard game.")
        get_tree().quit(EXIT_FAILURE)
        return

    # Loopback is for a server and guests running on this test machine.
    var started: Dictionary = await game.start_dedicated(TEST_BIND_ADDRESS).wait()
    if not started.ok:
        push_error(started.message)
        get_tree().quit(EXIT_FAILURE)
        return
    print("Cargo yard ready at %s:%d" % [TEST_BIND_ADDRESS, game_definition.multiplayer_rules.port])
```
