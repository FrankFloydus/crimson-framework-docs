## Example

```gdscript
extends Node

func run_start_dedicated_online(
    game_runtime: GameRuntime,
    public_address: String,
    bind_address: String,
    status_label: Label
) -> void:
    var result: GameOperation = game_runtime.start_dedicated_online(public_address, bind_address)
    if result == null:
        return
    var result_value: Dictionary = await result.wait()
    if not result_value.get("ok", false):
        status_label.text = str(result_value.get("message", "Operation failed."))
        return
    status_label.text = "Start dedicated online completed."
```
