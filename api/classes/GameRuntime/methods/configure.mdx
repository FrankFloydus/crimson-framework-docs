## Example

```gdscript
extends Node

const WORLD_ID: StringName = &"harbor"

@export var world_scene: PackedScene
@export var player_scene: PackedScene

# Assign a world with a spawn point and a Node3D player scene.
func _ready() -> void:
    if world_scene == null or player_scene == null:
        push_error("Assign the world and player scenes.")
        return

    var definition: GameDefinition = GameRuntime.configure()
    definition = definition.with_starting_world(WORLD_ID, world_scene)
    definition = definition.with_player(player_scene)
    var game: GameRuntime = definition.boot(self)
    if game == null:
        push_error("Could not attach the game.")
        return

    var started: Dictionary = await game.play_solo().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("Solo world ready: ", game.get_world_id())
```
