## Example

```gdscript
const LOCAL_ADDRESS: String = "127.0.0.1"

# The guest needs the same multiplayer port and a compatible world definition.
func join_local_host(game: GameRuntime) -> void:
    var result: Dictionary = await game.join_game(LOCAL_ADDRESS).wait()
    if not result.ok:
        push_error(result.message)
        return
    print("Joined: ", game.get_world().name)
```
