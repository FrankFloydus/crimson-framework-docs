## Description

Use the same definition and game node path in the host and guest builds.

## Example

```gdscript
extends Node

@export var game_definition: GameDefinition

func _ready() -> void:
    if game_definition == null:
        push_error("Assign the world and player scenes through a game definition.")
        return
    var game: GameRuntime = GameRuntime.new()
    # Set this before adding the runtime to the scene tree.
    game.definition = game_definition
    add_child(game)
    var started: Dictionary = await game.play_solo().wait()
    if not started.ok:
        push_error(started.message)
        return
    print("World ready: ", game.get_world_id())
```
