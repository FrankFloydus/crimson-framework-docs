## Example

```gdscript
extends Node

func run_get_value(steam_listing_result: SteamListingResult, name_label: Label) -> void:
    if steam_listing_result == null:
        return
    if not steam_listing_result.ok:
        push_warning(steam_listing_result.message)
        return
    var result: SteamListing = steam_listing_result.get_value()
    if result == null:
        return
    var name: String = result.name
    name_label.text = "Name: %s" % str(name)
```
