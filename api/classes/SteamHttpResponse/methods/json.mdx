## Example

```gdscript
extends Node

func run_json(steam_http_response: SteamHttpResponse, apply_json: Callable) -> void:
    var result: SteamJsonResult = steam_http_response.json()
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: Variant = result.value
    apply_json.call(payload)
```
