## Example

```gdscript
extends Node

func run_is_headset_streaming_enabled(steam_client: SteamClient, status_label: Label) -> void:
    if not steam_client.is_ready():
        return
    var steam_device: SteamDevice = steam_client.device()
    if steam_device == null:
        return
    var result: bool = steam_device.is_headset_streaming_enabled()
    status_label.text = "Headset streaming enabled: yes" if result else "Headset streaming enabled: no"
```
