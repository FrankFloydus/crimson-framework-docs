## Signature

```text
open_vr_dashboard() -> Error
```

## Return value

`Error`

## Description

`OK` acknowledges the SDK request. Returns `ERR_UNAVAILABLE` when Steam is outside VR mode or the scope or dashboard capability is unavailable.

## Example

```gdscript
extends Node

func run_open_vr_dashboard(steam_client: SteamClient, status_label: Label) -> void:
    if not steam_client.is_ready():
        return
    var steam_device: SteamDevice = steam_client.device()
    if steam_device == null:
        return
    var result: Error = steam_device.open_vr_dashboard()
    if result != OK:
        status_label.text = error_string(result)
        return
    status_label.text = "Open vr dashboard accepted."
```
