## Signature

```text
register_scene(scene_id: StringName, scene_path: String) -> Error
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `scene_id` | `StringName` | — |
| `scene_path` | `String` | — |

## Return value

`Error`

## Description

IDs are 1 to 64 ASCII characters, begin with a letter or digit, and can contain letters, digits, dots, hyphens, and underscores. Duplicate IDs return `ERR\_ALREADY\_EXISTS`.

## Example

```gdscript
extends Node

const TRAINING_ROOM_ID: StringName = &"training_room"

# Assign a persistent node for the mounted World and select a local scene.
@export var world_container: Node
@export_file("*.tscn", "*.scn") var training_room_path: String = ""

var _loader: SceneLoadManager

func _ready() -> void:
    if world_container == null or training_room_path.is_empty():
        push_error("Assign the world container and a local training-room scene.")
        return

    _loader = SceneLoadManager.new()
    add_child(_loader)
    var container_status: Error = _loader.set_world_container(world_container)
    if container_status != OK:
        push_error(error_string(container_status))
        return

    var registered: Error = _loader.register_scene(TRAINING_ROOM_ID, training_room_path)
    if registered != OK:
        push_error(error_string(registered))
        return

    _loader.load_completed.connect(_on_load_completed)
    _loader.load_failed.connect(_on_load_failed)
    var admitted: Error = _loader.request_load(TRAINING_ROOM_ID)
    if admitted != OK:
        push_warning(error_string(admitted))

func _on_load_completed(scene_id: StringName) -> void:
    var world: Node = _loader.get_current_scene()
    if world != null:
        print("Mounted ", scene_id, " under ", world.get_parent().name)

func _on_load_failed(_scene_id: StringName, _error: int, message: String) -> void:
    push_warning(message)
```
