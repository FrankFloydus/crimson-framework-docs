## Example

### Travel to the foundry when the lift arrives

Connect the lift's arrival event to `arrive_at_foundry()`. When the destination loads, the travel panel names the new world. [World catalog setup](/coordinated-lan#prepare-the-world-catalog)

```gdscript
extends Node

const TRAINING_ID: StringName = &"training"
const FOUNDRY_ID: StringName = &"foundry"

@export_file("*.tscn") var training_scene: String
@export_file("*.tscn") var foundry_scene: String
@export var travel_status: Label

var _game: Game
var _foundry: WorldDefinition

func _ready() -> void:
    var training: WorldDefinition = WorldDefinition.at(TRAINING_ID, training_scene)
    _foundry = WorldDefinition.at(FOUNDRY_ID, foundry_scene)
    _game = Game.campaign(training).with_world(_foundry).boot(self)
    var started: GameResult = await _game.play()
    if not started.ok:
        travel_status.text = started.message

func arrive_at_foundry() -> void:
    if _game.get_phase() != Game.RUNNING:
        return
    var scenes: GameScenes = _game.scenes()
    var arrived: GameResult = await scenes.load(_foundry)
    travel_status.text = "Arrived at %s." % arrived.world_id if arrived.ok else arrived.message
```
