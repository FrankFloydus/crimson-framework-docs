## Example

```gdscript
extends Node

func run_is_available(steam_client: SteamClient, action_button: Button) -> void:
    if not steam_client.is_ready():
        return
    var steam_http: SteamHttp = steam_client.http()
    if steam_http == null:
        return
    var result: bool = steam_http.is_available()
    action_button.disabled = not result
```
