## Example

### Answer a Steam invitation from the title screen

Show the exact invitation, then let the player join the training room or dismiss the offer. [Steam invitation setup](/steam/listen-servers#invite-and-join)

```gdscript
extends Node

@export var level: PackedScene
@export var player: PackedScene
@export var accept_button: Button
@export var dismiss_button: Button
@export var prompt: Label
@export var status: Label

var _game: Game
var _shown_offer: SteamInvite
var _joining: bool = false

func _ready() -> void:
    _update_buttons()
    _game = Game.configure(level, player).steam().boot(self)
    var startup: GameResult = _game.boot_result()
    if not startup.ok:
        status.text = startup.message
        return

    accept_button.pressed.connect(_accept_offer)
    dismiss_button.pressed.connect(_dismiss_offer)
    _game.invite_changed.connect(_show_offer)
    _show_offer(_game.pending_invite())

func _show_offer(offer: SteamInvite) -> void:
    _shown_offer = offer
    prompt.text = "Join %s's training room?" % offer.from_steam_id if offer != null else ""
    _update_buttons()

func _update_buttons() -> void:
    var disabled: bool = _shown_offer == null or _joining
    accept_button.disabled = disabled
    dismiss_button.disabled = disabled

func _accept_offer() -> void:
    var confirmed: SteamInvite = _shown_offer
    _joining = true
    _update_buttons()
    var joined: GameResult = await _game.accept_invite(confirmed)
    status.text = "Joined the training room." if joined.ok else joined.message
    _joining = false
    _update_buttons()

func _dismiss_offer() -> void:
    var dismissed: GameResult = _game.dismiss_invite(_shown_offer)
    status.text = "Invitation dismissed." if dismissed.ok else dismissed.message
```
