## Example

### Open the training room from the title screen

Press Play to mount the training room and give the player control. The menu hides after the local player spawns.

```gdscript
extends Node

const TRAINING_ROOM: StringName = &"training_room"

@export var level: PackedScene
@export var player: PackedScene
@export var title_menu: Control
@export var play_button: Button
@export var status: Label

var _game: Game

func _ready() -> void:
    play_button.disabled = true
    var setup: GameConfig = Game.configure(level, player).world_id(TRAINING_ROOM)
    _game = setup.boot(self)
    var booted: GameResult = _game.boot_result()
    if not booted.ok:
        status.text = booted.message
        return
    play_button.pressed.connect(_play)
    play_button.disabled = false

func _play() -> void:
    play_button.disabled = true
    var started: GameResult = await _game.play()
    if started.ok:
        title_menu.hide()
        return
    status.text = started.message
    play_button.disabled = false
```
