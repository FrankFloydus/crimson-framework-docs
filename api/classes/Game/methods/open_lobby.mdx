## Example

### Start a training match after a LAN guest joins

Run two local copies with the same level. Enable Host on one copy so the match starts after the other copy joins and both test identities are bound. [Coordinated LAN setup](/coordinated-lan#host-the-lobby-and-bind-players)

```gdscript
extends Node

const TRAINING_ID: StringName = &"training"
const LOOPBACK_ADDRESS: String = "127.0.0.1"
const HOST_PEER_ID: int = 1
const HOST_TEST_ID: String = "test_host"
const GUEST_TEST_ID: String = "test_guest"

@export_file("*.tscn") var training_scene: String
@export var host: bool = false

var _game: Game

func _ready() -> void:
    var training: WorldDefinition = WorldDefinition.at(TRAINING_ID, training_scene)
    _game = Game.campaign(training).lan().boot(self)
    if not _report_failure(_game.boot_result()):
        return
    _game.world_ready.connect(_on_world_ready)
    if host:
        if not _report_failure(_game.open_lobby(LOOPBACK_ADDRESS)):
            return
        if _game.bind_player(HOST_PEER_ID, HOST_TEST_ID) != OK:
            push_error("Could not bind the test host.")
            await _game.leave()
            return
        _game.peer_joined.connect(_start_with_guest)
        print("Lobby open. Start the guest now.")
    else:
        _report_failure(_game.join_lobby(LOOPBACK_ADDRESS))

func _start_with_guest(peer_id: int) -> void:
    # This loopback example reserves one identity for its test guest.
    if _game.bind_player(peer_id, GUEST_TEST_ID) != OK:
        push_error("Could not bind the test guest.")
        return
    var started: GameResult = await _game.start_match()
    _report_failure(started)

func _on_world_ready(world_id: StringName, _epoch: int) -> void:
    print("Match ready in ", world_id)

func _report_failure(result: GameResult) -> bool:
    if not result.ok:
        push_error(result.message)
    return result.ok
```
