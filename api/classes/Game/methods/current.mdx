## Example

### Choose the match for an external debug panel

Select the live match when both a preview and a match own game scopes. Clear the selection to return to automatic lookup. [Set up the debug panel](/game-services#select-a-game-from-a-debug-panel)

```gdscript
extends CheckButton

@export var match_owner: Node


func _ready() -> void:
    toggled.connect(_select_default)


func _select_default(use_match: bool) -> void:
    var selected: Game
    if use_match:
        selected = Game.for_node(match_owner)
        if selected == null:
            set_pressed_no_signal(false)
            push_warning("Start the match before selecting it.")
            return

    var status: int = Game.set_default(selected)
    if status != OK:
        set_pressed_no_signal(not use_match)
        push_error("Could not update the default game.")
        return

    var current: Game = Game.current()
    if current == null:
        print("No default game selected.")
    else:
        print("Debug game phase: %d" % current.get_phase())
```
