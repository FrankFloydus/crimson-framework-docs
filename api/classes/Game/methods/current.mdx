## Example

### Choose the match for an external debug panel

Choose the match when the preview and match both own booted scopes. Turn off the selection to return to automatic lookup. [Set up the debug panel](/game-services#select-a-game-from-a-debug-panel)

```gdscript
extends Node

const LEVEL: PackedScene = preload("res://scope_level.tscn")
const PLAYER: PackedScene = preload("res://scope_player.tscn")

@export var use_match_button: CheckButton

var _match_owner: Node


func _ready() -> void:
    var preview_owner: Node = Node.new()
    preview_owner.name = "Preview"
    add_child(preview_owner)

    _match_owner = Node.new()
    _match_owner.name = "Match"
    add_child(_match_owner)

    var config: GameConfig = Game.configure(LEVEL, PLAYER)
    var preview: Game = config.boot(preview_owner)
    var match_game: Game = config.boot(_match_owner)
    if not preview.boot_result().ok or not match_game.boot_result().ok:
        push_error("Could not attach both game scopes.")
        use_match_button.disabled = true
        return

    use_match_button.set_pressed_no_signal(false)
    use_match_button.toggled.connect(_select_default)


func _select_default(use_match: bool) -> void:
    var selected: Game
    if use_match:
        selected = Game.for_node(_match_owner)
        if selected == null:
            use_match_button.set_pressed_no_signal(false)
            push_warning("The match scope has retired.")
            return

    var status: int = Game.set_default(selected)
    if status != OK:
        use_match_button.set_pressed_no_signal(not use_match)
        push_error("Could not update the default game.")
        return

    var current: Game = Game.current()
    if current == null:
        print("No default game selected.")
    else:
        print("Match selected. Current phase: %d" % current.get_phase())
```
