## Example

### Host or join a four-player salvage run

The same menu can host the salvage yard or join another player's address. Both copies use the same scenes and UDP port.

```gdscript
extends Node

const COOP_SEATS: int = 4
const COOP_PORT: int = 24830

@export var salvage_yard: PackedScene
@export var player: PackedScene
@export var host_button: Button
@export var join_button: Button
@export var address: LineEdit
@export var status: Label

var _game: Game

func _ready() -> void:
    _set_busy(true)
    var setup: GameConfig = Game.configure(salvage_yard, player).lan(COOP_SEATS).on_port(COOP_PORT)
    _game = setup.boot(self)
    var booted: GameResult = _game.boot_result()
    if not booted.ok:
        status.text = booted.message
        return
    host_button.pressed.connect(_host)
    join_button.pressed.connect(_join)
    _set_busy(false)

func _host() -> void:
    _set_busy(true)
    status.text = "Opening the salvage yard…"
    var hosted: GameResult = await _game.host()
    _finish(hosted, "Hosting on UDP port %d." % _game.get_port())

func _join() -> void:
    if address.text.strip_edges().is_empty():
        status.text = "Enter the host's address."
        return
    _set_busy(true)
    status.text = "Joining the salvage yard…"
    var joined: GameResult = await _game.join(address.text.strip_edges())
    _finish(joined, "You joined the salvage run.")

func _finish(result: GameResult, success: String) -> void:
    status.text = success if result.ok else result.message
    _set_busy(result.ok)

func _set_busy(busy: bool) -> void:
    host_button.disabled = busy
    join_button.disabled = busy
```
