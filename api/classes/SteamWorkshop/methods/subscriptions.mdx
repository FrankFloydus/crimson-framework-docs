## Signature

```text
subscriptions(limit: int = 128, include_locally_disabled: bool = false) -> SteamWorkshopSubscriptionsResult
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `limit` | `int` | `128` |
| `include_locally_disabled` | `bool` | `false` |

## Return value

[`SteamWorkshopSubscriptionsResult`](/api/classes/SteamWorkshopSubscriptionsResult)

## Description

Set `include_locally_disabled` to include subscriptions disabled in the Steam client. The limit ranges from 1 to `MAX_SUBSCRIPTIONS`. Use a returned decimal ID with `item()` to inspect or install that item.

## Example

```gdscript
extends Node

const SUBSCRIPTION_LIMIT: int = 24

@export var steam_client: SteamClient
@export var subscription_list: ItemList
@export var status_label: Label

# Connect Refresh in your Workshop screen to this method.
func refresh_workshop_list() -> void:
    if steam_client == null or subscription_list == null or status_label == null:
        return
    var workshop: SteamWorkshop = steam_client.workshop()
    if workshop == null:
        status_label.text = "Workshop unavailable"
        return
    var result: SteamWorkshopSubscriptionsResult = workshop.subscriptions(SUBSCRIPTION_LIMIT)
    if not result.ok:
        status_label.text = result.message
        return

    var subscriptions: SteamWorkshopSubscriptions = result.value
    subscription_list.clear()
    for published_id: String in subscriptions.published_ids:
        subscription_list.add_item("Workshop item " + published_id)
    status_label.text = "Showing %d of %d subscribed items." % [
        subscription_list.item_count, subscriptions.total_count
    ]
```
