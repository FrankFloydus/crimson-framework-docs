## Signature

```text
query() -> SteamWorkshopQuery
```

## Return value

[`SteamWorkshopQuery`](/api/classes/SteamWorkshopQuery)

## Description

Use the query methods to branch filters before calling `SteamWorkshopQuery.fetch()`.

## Example

```gdscript
extends Node

func run_query(steam_client: SteamClient, apply_search_result: Callable) -> void:
    if not steam_client.is_ready():
        return
    var steam_workshop: SteamWorkshop = steam_client.workshop()
    if steam_workshop == null:
        return
    var result: SteamWorkshopQuery = steam_workshop.query()
    if result == null:
        return
    var search_result: SteamWorkshopPageResult = await result.fetch()
    if search_result == null:
        return
    if not search_result.ok:
        push_warning(search_result.message)
        return
    var payload: SteamWorkshopPage = search_result.value
    if payload == null:
        return
    apply_search_result.call(payload)
```
