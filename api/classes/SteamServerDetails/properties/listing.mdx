## Example

### Let the player pick a nearby arena server

Refresh to find open arenas below the ping limit, then select one and press Join. Launch the client with the same Steam app and game build as the servers.

```gdscript
extends Control

const ARENA_ID: StringName = &"arena"
const MAX_PING_MS: int = 120
const MIN_FREE_SEATS: int = 1

@export var arena: PackedScene
@export var player: PackedScene
@export var refresh_button: Button
@export var join_button: Button
@export var server_rows: ItemList
@export var status_label: Label

var _game: Game
var _browser: SteamServerBrowser
var _listings: Array[SteamListing] = []


func _ready() -> void:
    refresh_button.disabled = true
    join_button.disabled = true
    var setup: GameConfig = Game.configure(arena, player).world_id(ARENA_ID).steam()
    _game = setup.boot(self)
    var booted: GameResult = _game.boot_result()
    if not booted.ok:
        status_label.text = booted.message
        return
    var steam: Steamworks = _game.steam()
    if not steam.is_ready():
        status_label.text = steam.boot_result().message
        return
    _browser = steam.server_browser()
    if _browser == null or not _browser.is_available():
        status_label.text = "The Steam server browser is unavailable."
        return

    refresh_button.pressed.connect(_refresh)
    join_button.pressed.connect(_join_selected)
    server_rows.item_selected.connect(_on_server_selected)
    refresh_button.disabled = false
    status_label.text = "Refresh to find nearby arenas."


func _refresh() -> void:
    refresh_button.disabled = true
    join_button.disabled = true
    server_rows.clear()
    _listings.clear()
    status_label.text = "Finding nearby arenas…"

    var query: SteamServerQuery = _browser.query()
    query = query.on_map(ARENA_ID)
    query = query.with_room_for(MIN_FREE_SEATS).under_latency(MAX_PING_MS)
    var found: SteamListingsResult = await query.fetch()
    refresh_button.disabled = false
    if not found.ok:
        status_label.text = found.message
        return

    for listing: SteamListing in found.value:
        _listings.append(listing)
        server_rows.add_item("%s (%d ms)" % [listing.name, listing.ping_ms])
    if _listings.is_empty():
        status_label.text = "No nearby arenas with an open seat."
    else:
        status_label.text = "Select an arena to join."


func _on_server_selected(_row: int) -> void:
    join_button.disabled = refresh_button.disabled


func _join_selected() -> void:
    var selected_rows: PackedInt32Array = server_rows.get_selected_items()
    if selected_rows.is_empty():
        return
    refresh_button.disabled = true
    join_button.disabled = true
    status_label.text = "Checking the selected arena…"

    var selected: SteamListing = _listings[selected_rows[0]]
    var checked: SteamServerDetailsResult = await _browser.inspect(selected).wait()
    if not checked.ok:
        status_label.text = checked.message
        refresh_button.disabled = false
        join_button.disabled = false
        return

    status_label.text = "Joining %s (%d ms)…" % [
        checked.value.listing.name, checked.value.ping_ms
    ]
    var joined: GameResult = await _game.join_listing(checked.value.listing)
    if not joined.ok:
        status_label.text = joined.message
        refresh_button.disabled = false
        join_button.disabled = false
        return
    status_label.text = "Joined the arena."
```
