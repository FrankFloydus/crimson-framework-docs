## Example

```gdscript
extends Node

func run_get_value(
    steam_leaderboard_result: SteamLeaderboardResult,
    apply_value: Callable
) -> void:
    if steam_leaderboard_result == null:
        return
    if not steam_leaderboard_result.ok:
        push_warning(steam_leaderboard_result.message)
        return
    var result: SteamLeaderboard = steam_leaderboard_result.get_value()
    if result == null:
        return
    apply_value.call(result)
```
