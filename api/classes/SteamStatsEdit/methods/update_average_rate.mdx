## Description

Steam checks whether the named stat accepts average-rate updates when the draft is applied.

## Example

```gdscript
extends Node

func run_update_average_rate(
    steam_stats: SteamStats,
    name: String,
    count: float,
    seconds: float,
    confirmed: bool,
    apply_completion: Callable
) -> void:
    var steam_stats_edit: SteamStatsEdit = steam_stats.edit()
    if steam_stats_edit == null:
        return
    var result: SteamStatsEdit = steam_stats_edit.update_average_rate(name, count, seconds)
    if result == null:
        return
    if not confirmed:
        return
    var completion: SteamStatsSubmissionResult = result.submit()
    if completion == null:
        return
    if not completion.ok:
        push_warning(completion.message)
        return
    var payload: SteamStatsSubmission = completion.value
    if payload == null:
        return
    apply_completion.call(payload)
```
