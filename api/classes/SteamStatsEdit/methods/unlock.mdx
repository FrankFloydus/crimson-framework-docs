## Example

```gdscript
extends Node

func run_unlock(
    steam_stats: SteamStats,
    name: String,
    confirmed: bool,
    apply_completion: Callable
) -> void:
    var steam_stats_edit: SteamStatsEdit = steam_stats.edit()
    if steam_stats_edit == null:
        return
    var result: SteamStatsEdit = steam_stats_edit.unlock(name)
    if result == null:
        return
    if not confirmed:
        return
    var completion: SteamStatsSubmissionResult = result.submit()
    if completion == null:
        return
    if not completion.ok:
        push_warning(completion.message)
        return
    var payload: SteamStatsSubmission = completion.value
    if payload == null:
        return
    apply_completion.call(payload)
```
