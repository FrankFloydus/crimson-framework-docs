## Description

Return the verified backend state. The hint alone does not grant an entitlement.

## Example

```gdscript
extends SteamCommerceBackend

# Reconcile the hint with the current authenticated order on your game backend.
var refresh_after_hint: Callable

func _observe_authorization(order_id: String, authorized: bool, request: SteamCommerceRequest) -> void:
    var result: SteamPurchaseOutcomeResult
    if refresh_after_hint.is_valid():
        result = await refresh_after_hint.call(order_id, authorized) as SteamPurchaseOutcomeResult
    if result == null:
        result = SteamPurchaseOutcomeResult.create(ERR_UNAVAILABLE)
    request.complete_order(result)
```
