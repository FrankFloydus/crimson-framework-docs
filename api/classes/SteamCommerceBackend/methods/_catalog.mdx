## Example

```gdscript
extends SteamCommerceBackend

# The callback fetches the authenticated player's current server-priced offers.
var fetch_catalog: Callable

func _catalog(request: SteamCommerceRequest) -> void:
    var result: SteamPurchaseCatalogResult
    if fetch_catalog.is_valid():
        result = await fetch_catalog.call() as SteamPurchaseCatalogResult
    if result == null:
        result = SteamPurchaseCatalogResult.create(ERR_UNAVAILABLE)
    request.complete_catalog(result)
```
