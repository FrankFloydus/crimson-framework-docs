## Signature

```text
start_offline() -> Error
```

## Return value

`Error`

## Description

Return OK when running or an immediate Godot Error.

## Example

```gdscript
extends Node

const SPAWN_POINTS_PATH: NodePath = ^"SpawnPoints"
const CAMERA_PATH: NodePath = ^"Camera3D"

# Assign a Node3D world with a Marker3D under SpawnPoints and a
# Node3D player scene with a Camera3D at CAMERA_PATH.
@export var world_scene: PackedScene
@export var player_scene: PackedScene

func _ready() -> void:
    if world_scene == null or player_scene == null:
        push_error("Assign the world and player scenes.")
        return

    var definition: WorldSessionDefinition = WorldSession.configure()
    definition = definition.with_world(world_scene)
    definition = definition.with_players(player_scene, SPAWN_POINTS_PATH)
    definition = definition.with_local_view(CAMERA_PATH)
    var session: WorldSession = definition.create()
    add_child(session)

    var started: Error = session.start_offline()
    if started != OK:
        push_error("Could not open the local world. " + error_string(started))
        return

    var player: Node3D = session.get_local_player()
    if player != null:
        print("Local player spawned at ", player.global_position)
```
