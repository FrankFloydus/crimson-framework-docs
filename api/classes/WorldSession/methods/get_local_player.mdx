`Node3D`

## Description

## Example

```gdscript
extends Node

func run_get_local_player(world_session: WorldSession, local_player_label: Label) -> void:
    var result: Node3D = world_session.get_local_player()
    if result == null:
        return
    local_player_label.text = str(result.name)
```
