## Signature

```text
host() -> Error
```

## Return value

`Error`

## Description

The host uses the first marker and leaves the remaining markers for guests. Return OK when running or an immediate Godot Error.

## Example

```gdscript
extends Node

const GAME_PORT: int = 7000
const SPAWN_POINTS_PATH: NodePath = ^"SpawnPoints"
const CAMERA_PATH: NodePath = ^"Camera3D"

# Use the same world and player scenes on host and guest.
# The direct Marker3D children of SpawnPoints provide the available slots.
@export var world_scene: PackedScene
@export var player_scene: PackedScene
@export var host_button: Button

var _session: WorldSession

func _ready() -> void:
    if world_scene == null or player_scene == null or host_button == null:
        push_error("Assign the world, player scene, and Host button.")
        return

    var definition: WorldSessionDefinition = WorldSession.configure()
    definition = definition.with_world(world_scene)
    definition = definition.with_players(player_scene, SPAWN_POINTS_PATH)
    definition = definition.with_local_view(CAMERA_PATH).on_port(GAME_PORT)
    _session = definition.create()
    add_child(_session)
    host_button.pressed.connect(_host_world)

func _host_world() -> void:
    var started: Error = _session.host()
    if started != OK:
        push_warning("Could not host the world. " + error_string(started))
        return

    host_button.disabled = true
    print("Guests can join this host on UDP port ", _session.get_port())
```
