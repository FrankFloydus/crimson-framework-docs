## Signature

```text
join(address: String) -> Error
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `address` | `String` | — |

## Return value

`Error`

## Description

OK means the attempt started. Wait for RUNNING before accessing the local player.

## Example

```gdscript
extends Node

const GAME_PORT: int = 7000
const SPAWN_POINTS_PATH: NodePath = ^"SpawnPoints"
const CAMERA_PATH: NodePath = ^"Camera3D"

# Assign the same scenes as the host and enter its reachable address.
@export var world_scene: PackedScene
@export var player_scene: PackedScene
@export var host_address: String = "127.0.0.1"

var _session: WorldSession

func _ready() -> void:
    if world_scene == null or player_scene == null:
        push_error("Assign the world and player scenes used by the host.")
        return

    var definition: WorldSessionDefinition = WorldSession.configure()
    definition = definition.with_world(world_scene)
    definition = definition.with_players(player_scene, SPAWN_POINTS_PATH)
    definition = definition.with_local_view(CAMERA_PATH).on_port(GAME_PORT)
    _session = definition.create()
    add_child(_session)
    _session.state_changed.connect(_on_state_changed)
    _session.session_ended.connect(_on_session_ended)

    var started: Error = _session.join(host_address)
    if started != OK:
        push_error("Connection could not start. " + error_string(started))

func _on_state_changed(state: int) -> void:
    if state != WorldSession.RUNNING:
        return
    var player: Node3D = _session.get_local_player()
    if player != null:
        print("Joined the world as ", player.name)

func _on_session_ended(reason: StringName) -> void:
    push_warning("Session ended. " + String(reason))
```
