## Example

```gdscript
extends Node

func run_is_available(steam_client: SteamClient, action_button: Button) -> void:
    if not steam_client.is_ready():
        return
    var steam_friends: SteamFriends = steam_client.friends()
    if steam_friends == null:
        return
    var result: bool = steam_friends.is_available()
    action_button.disabled = not result
```
