## Example

```gdscript
extends Node

func run_rich_presence(
    steam_client: SteamClient,
    key: String,
    rich_presence_label: Label
) -> void:
    if not steam_client.is_ready():
        return
    var steam_friends: SteamFriends = steam_client.friends()
    if steam_friends == null:
        return
    var result: String = steam_friends.rich_presence(key)
    rich_presence_label.text = "Rich presence: %s" % str(result)
```
