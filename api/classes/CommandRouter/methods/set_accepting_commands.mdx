## Signature

```text
set_accepting_commands(enabled: bool) -> Error
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `enabled` | `bool` | — |

## Return value

`Error`

## Description

Returns `ERR\_UNCONFIGURED` before configuration or `ERR\_BUSY` during dispatch or from another thread.

## Example

### Control a repair drone's work light

Press **Toggle Work Light** while controlling the pilot to see the command rejected. Switch to the repair drone and press it again to turn on the light.

```gdscript
extends Node

const MATCH_ID: String = "hangar_training"
const HANGAR_EPOCH: int = 1
const PLAYERS: String = "players"
const CHARACTERS: String = "characters"
const LOCAL_PLAYER: String = "trainee"
const PILOT: String = "pilot"
const DRONE: String = "repair_drone"
const TOGGLE_LIGHT: String = "toggle_work_light"

@export var pilot_button: Button
@export var drone_button: Button
@export var light_button: Button
@export var drone_light: SpotLight3D
@export var status: Label

var _stores: TransactionCoordinator = TransactionCoordinator.new()
var _possession: PlayerPossession = PlayerPossession.new()
var _router: CommandRouter = CommandRouter.new()
var _sequence: int = 0


func _ready() -> void:
    drone_light.visible = false
    var error: Error = _configure_training()
    if error != OK:
        status.text = error_string(error)
        return
    if not _switch_to(PILOT):
        return
    pilot_button.pressed.connect(_switch_to.bind(PILOT))
    drone_button.pressed.connect(_switch_to.bind(DRONE))
    light_button.pressed.connect(_request_light_toggle)


func _configure_training() -> Error:
    var error: Error = _stores.register_store(PLAYERS, {LOCAL_PLAYER: {}}, _keep_roster)
    if error != OK:
        return error
    error = _stores.register_store(CHARACTERS, {PILOT: {}, DRONE: {}}, _keep_roster)
    if error != OK:
        return error
    error = _possession.configure(_stores, PLAYERS, CHARACTERS)
    if error != OK:
        return error
    error = _router.configure(MATCH_ID, HANGAR_EPOCH, _possession)
    if error != OK:
        return error
    error = _router.register_handler(TOGGLE_LIGHT, _toggle_light, true)
    if error != OK:
        return error
    return _router.set_accepting_commands(true)


func _keep_roster(current: Dictionary, _operations: Array, _context: Dictionary) -> Dictionary:
    return {"ok": true, "state": current}


func _switch_to(character: String) -> bool:
    var revision: int = _possession.get_revision(LOCAL_PLAYER)
    var result: Dictionary = _possession.try_set_primary(LOCAL_PLAYER, character, revision)
    if result.ok:
        status.text = "Controlling the repair drone." if character == DRONE else "Controlling the pilot."
    else:
        status.text = result.message
    return result.ok


func _request_light_toggle() -> void:
    _sequence += 1
    var result: Dictionary = _router.submit(LOCAL_PLAYER, {
        "match_id": MATCH_ID,
        "world_epoch": HANGAR_EPOCH,
        "sequence": _sequence,
        "type": TOGGLE_LIGHT,
        "payload": {},
        "character_id": _possession.get_primary(LOCAL_PLAYER),
        "possession_revision": _possession.get_revision(LOCAL_PLAYER),
    })
    status.text = result.message


func _toggle_light(_player_id: String, intent: Dictionary) -> Dictionary:
    if intent.character_id != DRONE:
        return {"ok": false, "message": "Take control of the drone to use its work light."}
    drone_light.visible = not drone_light.visible
    return {"ok": true, "message": "Drone light on." if drone_light.visible else "Drone light off."}
```
