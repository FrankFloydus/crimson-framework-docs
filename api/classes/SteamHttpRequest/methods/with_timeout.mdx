extends Node

func run_with_timeout(
    steam_http_request: SteamHttpRequest,
    milliseconds: int,
    apply_send: Callable

    var result: SteamHttpRequest = steam_http_request.with_timeout(milliseconds)
    if result == null:
        return
    var completion: SteamHttpResponseOperation = result.send()
    if completion == null:
        return
    var result_value: SteamHttpResponseResult = await completion.wait()
    if result_value == null:
        return
    if not result_value.ok:
        push_warning(result_value.message)
        return
    var payload: SteamHttpResponse = result_value.value
    if payload == null:
        return
    apply_send.call(payload)
