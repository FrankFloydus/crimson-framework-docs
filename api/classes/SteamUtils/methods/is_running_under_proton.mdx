## Description

Returns `false` when the query is unavailable.

## Example

```gdscript
extends Node

func run_is_running_under_proton(steam_client: SteamClient, status_label: Label) -> void:
    if not steam_client.is_ready():
        return
    var steam_utils: SteamUtils = steam_client.utils()
    if steam_utils == null:
        return
    var result: bool = steam_utils.is_running_under_proton()
    status_label.text = "Running under proton: yes" if result else "Running under proton: no"
```
