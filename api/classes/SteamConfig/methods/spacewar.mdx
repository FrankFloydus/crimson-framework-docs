## Description

Supply the same 1 to 64 character group on both peers. Groups allow ASCII letters, digits, underscores, and hyphens. A group separates discovery and join for games sharing the development AppID. The process must already launch in the Spacewar context.

## Example

### Find an open Spacewar arena

Press Find Game to join an open test arena. Launch both copies in Spacewar with the same test group and have one copy host first. [Launch with an AppID](/steam/setup#launch-with-an-appid)

```gdscript
extends Node

const TEST_GROUP: String = "crimson-arena-test"
const ARENA_ID: StringName = &"arena"

@export var arena: PackedScene
@export var player: PackedScene
@export var find_game: Button
@export var status: Label

var _game: Game

func _ready() -> void:
    find_game.disabled = true
    var settings: SteamConfig = SteamConfig.spacewar(TEST_GROUP).public_games()
    var setup: GameConfig = Game.configure(arena, player).world_id(ARENA_ID).steam(settings)
    _game = setup.boot(self)
    var booted: GameResult = _game.boot_result()
    if not booted.ok:
        status.text = booted.message
        return
    find_game.pressed.connect(_find_open_arena)
    find_game.disabled = false

func _find_open_arena() -> void:
    find_game.disabled = true
    status.text = "Finding an open arena…"
    var joined: GameResult = await _game.find_and_join()
    status.text = "You joined the arena." if joined.ok else joined.message
    find_game.disabled = joined.ok
```
