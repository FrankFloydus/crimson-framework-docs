## Example

```gdscript
extends Node

func run_wait(
    steam_item_definitions_operation: SteamItemDefinitionsOperation,
    results_list: ItemList
) -> void:
    var result: SteamItemDefinitionsResult = await steam_item_definitions_operation.wait()
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: Array[SteamItemDefinition] = result.value
    results_list.clear()
    for entry: SteamItemDefinition in payload:
        results_list.add_item(str(entry.display_name))
```
