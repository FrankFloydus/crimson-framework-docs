## Signature

```text
catalog() -> Variant
```

## Return value

`Variant`

Await the call to receive [`SteamPurchaseCatalogResult`](/api/classes/SteamPurchaseCatalogResult).

## Example

```gdscript
extends Node

@export var game: GameRuntime
@export var shop_status: Label

# Use a game already attached with SteamProvider.with_commerce_backend().
func refresh_attached_shop() -> void:
    if game == null or shop_status == null:
        return
    var scope: SteamScope = SteamServices.for_game(game)
    if scope == null or not scope.is_ready():
        shop_status.text = "Steam is unavailable for this game."
        return
    var shop: SteamPurchases = scope.purchases()
    if shop == null:
        shop_status.text = "The shop service is unavailable."
        return
    var catalog: SteamPurchaseCatalogResult = await shop.catalog()
    if not catalog.ok:
        shop_status.text = catalog.message
        return
    shop_status.text = "%d offers for this player" % catalog.value.entries.size()
```
