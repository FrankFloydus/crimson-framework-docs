## Example

```gdscript
extends Node

func run_beacons(steam_client: SteamClient, add_beacons_entry: Callable) -> void:
    if not steam_client.is_ready():
        return
    var steam_parties: SteamParties = steam_client.parties()
    if steam_parties == null:
        return
    var result: SteamPartyListingsResult = steam_parties.beacons()
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: Array[SteamPartyListing] = result.value
    for entry: SteamPartyListing in payload:
        add_beacons_entry.call(entry)
```
