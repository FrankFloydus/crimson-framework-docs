## Example

```gdscript
extends Node

const PACK_SIZE: Vector2i = Vector2i(4, 4)
const CHEST_SIZE: Vector2i = Vector2i(6, 4)
const MAP_ID: StringName = &"ruins_map"
const MAP_SIZE: Vector2i = Vector2i(2, 1)

# On the camp storage screen, assign a Stash Map button and a label
# next to the map's entry in the inventory list.
@export var stash_button: Button
@export var map_location: Label

var _pack: SpatialInventory = SpatialInventory.new()
var _party_chest: SpatialInventory = SpatialInventory.new()
var _quest_chest: SpatialInventory = SpatialInventory.new()
var _actions: SpatialInventoryActions = SpatialInventoryActions.new()

func _ready() -> void:
    if stash_button == null or map_location == null:
        push_error("Assign the Stash Map button and map location label.")
        return
    if _pack.configure(PACK_SIZE) != OK:
        push_error("Could not prepare the backpack.")
        return
    if _party_chest.configure(CHEST_SIZE) != OK or _quest_chest.configure(CHEST_SIZE) != OK:
        push_error("Could not prepare the camp chests.")
        return
    if _pack.try_place(MAP_ID, MAP_SIZE, Vector2i.ZERO) != OK:
        push_error("Could not give the player the ruins map.")
        return

    # Keep quest storage separate even if it appears first in the menu.
    _actions.destination_policy = _reject_quest_chest
    map_location.text = "Ruins map: Backpack"
    stash_button.pressed.connect(_stash_map)

func _stash_map() -> void:
    var result: Dictionary = _actions.quick_transfer(_pack, MAP_ID, [_quest_chest, _party_chest])
    if not result.get("completed", false):
        map_location.text = String(result.get("reason", "The map stayed in the backpack."))
        return
    map_location.text = "Ruins map: Party chest"
    stash_button.disabled = true

func _reject_quest_chest(context: Dictionary) -> String:
    if context["destination"] == _quest_chest:
        return "Quest storage only accepts active quest items."
    return ""
```
