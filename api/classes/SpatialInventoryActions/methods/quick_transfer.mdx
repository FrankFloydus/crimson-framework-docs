## Signature

```text
quick_transfer(source: SpatialInventory, id: StringName, destinations: Array = []) -> Dictionary
```

## Parameters

| Parameter | Type | Default |
| --- | --- | --- |
| `source` | [`SpatialInventory`](/api/classes/SpatialInventory) | — |
| `id` | `StringName` | — |
| `destinations` | `Array` | `[]` |

## Return value

`Dictionary`

## Description

Merge compatible stacks first when an adapter is available, then try to place the remaining item. Inspect `completed` and `remaining` because a partial transfer can return `OK`.

## Example

```gdscript
extends Node

const PACK_SIZE: Vector2i = Vector2i(4, 4)
const BELT_SIZE: Vector2i = Vector2i(1, 1)
const CHEST_SIZE: Vector2i = Vector2i(6, 4)
const ITEM: StringName = &"rolled_map_01"
const FOOTPRINT: Vector2i = Vector2i(2, 1)

func _ready() -> void:
    var pack: SpatialInventory = SpatialInventory.new()
    var belt: SpatialInventory = SpatialInventory.new()
    var chest: SpatialInventory = SpatialInventory.new()
    if pack.configure(PACK_SIZE) != OK or belt.configure(BELT_SIZE) != OK or chest.configure(CHEST_SIZE) != OK:
        push_error("Could not set up the inventory grids.")
        return
    if pack.try_place(ITEM, FOOTPRINT, Vector2i.ZERO) != OK:
        push_error("Could not place the map in the source pack.")
        return

    # A quick-move button tries the belt first and the nearby chest second.
    var actions: SpatialInventoryActions = SpatialInventoryActions.new()
    var destinations: Array = [belt, chest]
    var transfer: Dictionary = actions.quick_transfer(pack, ITEM, destinations)
    if not transfer.get("completed", false):
        push_warning(String(transfer.get("reason", "Could not transfer the map.")))
        return

    var placement: Dictionary = chest.get_placement(ITEM)
    print("Refresh the chest at ", placement["origin"], " and clear the old pack cell.")
```
