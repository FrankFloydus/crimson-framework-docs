## Example

```gdscript
extends Node

const PACK_SIZE: Vector2i = Vector2i(4, 4)
const CHEST_SIZE: Vector2i = Vector2i(6, 4)
const LOOT_SIZE: Vector2i = Vector2i(2, 1)
const MAP_ID: StringName = &"ruins_map"

# Assign the Take Map button and its inventory row's location label.
@export var take_map_button: Button
@export var map_location: Label

var _loot: SpatialInventory = SpatialInventory.new()
var _pack: SpatialInventory = SpatialInventory.new()
var _cart: SpatialInventory = SpatialInventory.new()
var _actions: SpatialInventoryActions = SpatialInventoryActions.new()

func _ready() -> void:
    if take_map_button == null or map_location == null:
        push_error("Assign the Take Map button and map location label.")
        return
    if _loot.configure(LOOT_SIZE) != OK or _pack.configure(PACK_SIZE) != OK:
        push_error("Could not set up the loot and backpack inventories.")
        return
    if _cart.configure(CHEST_SIZE) != OK or _loot.try_place(MAP_ID, LOOT_SIZE, Vector2i.ZERO) != OK:
        push_error("Could not set up the supply cart and ruins map.")
        return

    _actions.set_destination_provider(_loot_destinations)
    map_location.text = "Ruins map: Ruins chest"
    take_map_button.pressed.connect(_take_map)

func _take_map() -> void:
    # With no explicit destinations, the provider chooses where the loot goes.
    var result: Dictionary = _actions.quick_transfer(_loot, MAP_ID)
    if not result.get("completed", false):
        map_location.text = String(result.get("reason", "No room for the ruins map."))
        return
    var stashed_in_pack: bool = not _pack.get_placement(MAP_ID).is_empty()
    map_location.text = "Ruins map: Backpack" if stashed_in_pack else "Ruins map: Supply cart"
    take_map_button.disabled = true

func _loot_destinations(_context: Dictionary) -> Array:
    # Keep carried loot within reach and send overflow to the party's cart.
    return [_pack, _cart]
```
