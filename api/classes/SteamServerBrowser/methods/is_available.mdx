## Example

```gdscript
extends Node

func run_is_available(steam_client: SteamClient, action_button: Button) -> void:
    if not steam_client.is_ready():
        return
    var steam_server_browser: SteamServerBrowser = steam_client.server_browser()
    if steam_server_browser == null:
        return
    var result: bool = steam_server_browser.is_available()
    action_button.disabled = not result
```
