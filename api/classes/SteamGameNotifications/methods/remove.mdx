## Example

```gdscript
extends Node

func run_remove(
    steam_publisher: SteamPublisher,
    session_id: String,
    operation_key: String,
    status_label: Label
) -> void:
    if not steam_publisher.is_ready():
        return
    var steam_game_notifications: SteamGameNotifications = steam_publisher.game_notifications()
    if steam_game_notifications == null:
        return
    var result: SteamNotificationBoolOperation = steam_game_notifications.remove(session_id, operation_key)
    if result == null:
        return
    var result_value: SteamBoolResult = await result.wait()
    if result_value == null:
        return
    if not result_value.ok:
        push_warning(result_value.message)
        return
    var payload: bool = result_value.value
    status_label.text = "Remove: yes" if payload else "Remove: no"
```
