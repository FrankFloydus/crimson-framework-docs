## Example

```gdscript
extends Node

func run_get_value(
    steam_app_ownership_result: SteamAppOwnershipResult,
    app_id_label: Label
) -> void:
    if steam_app_ownership_result == null:
        return
    if not steam_app_ownership_result.ok:
        push_warning(steam_app_ownership_result.message)
        return
    var result: SteamAppOwnership = steam_app_ownership_result.get_value()
    if result == null:
        return
    var app_id: int = result.app_id
    app_id_label.text = "App id: %s" % str(app_id)
```
