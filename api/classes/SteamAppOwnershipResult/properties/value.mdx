func refresh_value(
    steam_app_ownership_result: SteamAppOwnershipResult,
    app_id_label: Label
) -> void:
    if steam_app_ownership_result == null:
        return
    if not steam_app_ownership_result.ok:
        push_warning(steam_app_ownership_result.message)
        return
    var value: SteamAppOwnership = steam_app_ownership_result.value
    if value == null:
        return
    var app_id: int = value.app_id
