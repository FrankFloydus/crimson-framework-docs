## Description

Steam allows one pending request across client scopes and enforces its own rate limit.

## Example

```gdscript
extends Node

func run_ownership_ticket(
    steam_client: SteamClient,
    challenge: PackedByteArray,
    apply_ownership_ticket: Callable
) -> void:
    if not steam_client.is_ready():
        return
    var steam_auth: SteamAuth = steam_client.auth()
    if steam_auth == null:
        return
    var result: SteamOwnershipTicketResult = await steam_auth.ownership_ticket(challenge)
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: SteamOwnershipTicket = result.value
    if payload == null:
        return
    apply_ownership_ticket.call(payload)
```
