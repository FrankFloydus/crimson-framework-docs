## Example

```gdscript
extends Node

func run_input(steam_scope: SteamScope, add_controllers_entry: Callable) -> void:
    var result: SteamInput = steam_scope.input()
    if result == null:
        return
    var controllers: Array[SteamController] = result.controllers()
    for entry: SteamController in controllers:
        add_controllers_entry.call(entry)
```
