## Signature

```text
workshop() -> SteamWorkshop
```

## Return value

[`SteamWorkshop`](/api/classes/SteamWorkshop)

## Description

Returns null after the attachment ends.

## Example

```gdscript
extends Node

func run_workshop(steam_scope: SteamScope, apply_workshop_results: Callable) -> void:
    var result: SteamWorkshop = steam_scope.workshop()
    if result == null:
        return
    var workshop_query: SteamWorkshopQuery = result.query()
    if workshop_query == null:
        return
    var workshop_results: SteamWorkshopPageResult = await workshop_query.fetch()
    if workshop_results == null:
        return
    if not workshop_results.ok:
        push_warning(workshop_results.message)
        return
    var payload: SteamWorkshopPage = workshop_results.value
    if payload == null:
        return
    apply_workshop_results.call(payload)
```
