## Example

```gdscript
extends Node

func run_auth(
    steam_scope: SteamScope,
    audience: String,
    verify_ticket_on_backend: Callable
) -> void:
    var result: SteamAuth = steam_scope.auth()
    if result == null:
        return
    var web_ticket: SteamWebTicketResult = await result.web_ticket(audience)
    if web_ticket == null:
        return
    if not web_ticket.ok:
        push_warning(web_ticket.message)
        return
    var payload: SteamWebTicket = web_ticket.value
    if payload == null:
        return
    var ticket_bytes: PackedByteArray = payload.bytes()
    if ticket_bytes.is_empty():
        payload.close()
        return
    # This callback sends bytes over your authenticated backend connection.
    var verified: bool = await verify_ticket_on_backend.call(ticket_bytes)
    payload.close()
    if not verified:
        push_warning("The backend did not verify the Steam ticket.")
```
