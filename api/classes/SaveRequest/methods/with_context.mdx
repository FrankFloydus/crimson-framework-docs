## Description

Use small fields such as a location or chapter label. Persist gameplay state in the match stores.

## Example

```gdscript
extends Node

func run_with_context(
    save_request: SaveRequest,
    context: Dictionary,
    confirmed: bool,
    status_label: Label
) -> void:
    var result: SaveRequest = save_request.with_context(context)
    if result == null:
        return
    if not confirmed:
        return
    var completion: GameOperation = result.submit()
    if completion == null:
        return
    var result_value: Dictionary = await completion.wait()
    if not result_value.get("ok", false):
        status_label.text = str(result_value.get("message", "Operation failed."))
        return
    status_label.text = "Submit completed."
```
