## Description

Submission sanitizes and bounds the label stored in the generation.

## Example

```gdscript
extends Node

func run_named(
    save_request: SaveRequest,
    label: String,
    confirmed: bool,
    status_label: Label
) -> void:
    var result: SaveRequest = save_request.named(label)
    if result == null:
        return
    if not confirmed:
        return
    var completion: GameOperation = result.submit()
    if completion == null:
        return
    var result_value: Dictionary = await completion.wait()
    if not result_value.get("ok", false):
        status_label.text = str(result_value.get("message", "Operation failed."))
        return
    status_label.text = "Submit completed."
```
