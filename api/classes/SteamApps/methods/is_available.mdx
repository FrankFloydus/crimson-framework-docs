## Example

```gdscript
extends Node

func run_is_available(steam_client: SteamClient, action_button: Button) -> void:
    if not steam_client.is_ready():
        return
    var steam_apps: SteamApps = steam_client.apps()
    if steam_apps == null:
        return
    var result: bool = steam_apps.is_available()
    action_button.disabled = not result
```
