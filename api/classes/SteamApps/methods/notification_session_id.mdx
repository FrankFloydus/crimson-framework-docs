## Signature

```text
notification_session_id() -> String
```

## Return value

`String`

## Description

Returns an empty string when absent, invalid, or unavailable. Ask the game's backend to authorize the selected session for the authenticated player before loading it.

## Example

```gdscript
extends Node

func run_notification_session_id(
    steam_client: SteamClient,
    notification_session_id_label: Label
) -> void:
    if not steam_client.is_ready():
        return
    var steam_apps: SteamApps = steam_client.apps()
    if steam_apps == null:
        return
    var result: String = steam_apps.notification_session_id()
    notification_session_id_label.text = "Notification session id: %s" % str(result)
```
