## Example

```gdscript
extends Node

func run_is_dlc_installed(steam_client: SteamClient, app_id: int, status_label: Label) -> void:
    if not steam_client.is_ready():
        return
    var steam_apps: SteamApps = steam_client.apps()
    if steam_apps == null:
        return
    var result: bool = steam_apps.is_dlc_installed(app_id)
    status_label.text = "Dlc installed: yes" if result else "Dlc installed: no"
```
