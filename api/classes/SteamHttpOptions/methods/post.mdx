## Example

```gdscript
extends Node

func run_post(
    steam_http_options: SteamHttpOptions,
    url: String,
    data: Dictionary,
    apply_post: Callable
) -> void:
    var result: SteamHttpResponseResult = await steam_http_options.post(url, data)
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: SteamHttpResponse = result.value
    if payload == null:
        return
    apply_post.call(payload)
```
