## Signature

```text
unsubscribe() -> Variant
```

## Return value

`Variant`

Await the call to receive [`SteamWorkshopSubscriptionResult`](/api/classes/SteamWorkshopSubscriptionResult).

## Description

Steam may remove the installed files after the game closes.

## Example

```gdscript
extends Node

func run_unsubscribe(
    steam_workshop_item_ref: SteamWorkshopItemRef,
    published_id_label: Label
) -> void:
    var result: SteamWorkshopSubscriptionResult = await steam_workshop_item_ref.unsubscribe()
    if result == null:
        return
    if not result.ok:
        push_warning(result.message)
        return
    var payload: SteamWorkshopSubscription = result.value
    if payload == null:
        return
    var published_id: String = payload.published_id
    published_id_label.text = "Published id: %s" % str(published_id)
```
