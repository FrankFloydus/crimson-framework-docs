## Example

```gdscript
extends Node

func run_open_profile(steam_client: SteamClient, steam_id: String, status_label: Label) -> void:
    if not steam_client.is_ready():
        return
    var steam_overlay: SteamOverlay = steam_client.overlay()
    if steam_overlay == null:
        return
    var result: Error = steam_overlay.open_profile(steam_id)
    if result != OK:
        status_label.text = error_string(result)
        return
    status_label.text = "Open profile accepted."
```
