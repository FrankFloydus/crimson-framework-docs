## Description

The container's ordered Marker3D children define the legacy spawn slots.

## Example

```gdscript
extends Node

const ARRIVAL_MARKERS: NodePath = ^"Landing/Arrivals"
const CAMERA_PATH: NodePath = ^"Camera3D"

# In the world scene, place the Marker3D spawn children directly under
# Landing/Arrivals. Assign a player scene with a Camera3D at CAMERA_PATH.
@export var world_scene: PackedScene
@export var player_scene: PackedScene

func _ready() -> void:
    if world_scene == null or player_scene == null:
        push_error("Assign the world and player scenes.")
        return

    var definition: WorldSessionDefinition = WorldSession.configure()
    definition = definition.with_world(world_scene).with_players(player_scene)
    definition = definition.with_local_view(CAMERA_PATH)
    definition.spawn_points_path = ARRIVAL_MARKERS
    var session: WorldSession = definition.create()
    add_child(session)

    var started: Error = session.start_offline()
    if started != OK:
        push_error(error_string(started))
        return

    var player: Node3D = session.get_local_player()
    if player != null:
        print("Arrived at the first authored marker at ", player.global_position)
```
