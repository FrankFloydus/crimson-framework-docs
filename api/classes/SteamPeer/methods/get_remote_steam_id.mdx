## Example

```gdscript
extends Node

const REMOTE_STEAM_ID_METHOD: StringName = &"get_remote_steam_id"

# On the host's moderation panel, assign its attached GameSession and
# the roster ItemList. The session must use the native Steam transport.
@export var session: GameSession
@export var guest_roster: ItemList

func _ready() -> void:
    if session == null or guest_roster == null:
        push_error("Assign the Steam session and its moderation roster.")
        return
    session.peer_joined.connect(_on_peer_joined)
    session.peer_left.connect(_on_peer_left)
    session.session_ended.connect(_on_session_ended)
    for peer_id: int in session.get_connected_peer_ids():
        _on_peer_joined(peer_id)

func _on_peer_joined(peer_id: int) -> void:
    if session.get_state() != GameSession.HOSTING:
        return
    var peer: MultiplayerPeer = session.multiplayer.multiplayer_peer
    # Crimson creates SteamPeer internally and exposes it through MultiplayerPeer.
    if peer == null or not peer.has_method(REMOTE_STEAM_ID_METHOD):
        return
    var account_id: int = int(peer.call(REMOTE_STEAM_ID_METHOD, peer_id))
    if account_id == 0:
        return
    # The roster identifies the Steam account even if the player changes their nickname.
    var row: int = guest_roster.add_item("Steam account %d" % account_id)
    guest_roster.set_item_metadata(row, peer_id)

func _on_peer_left(peer_id: int) -> void:
    for row: int in range(guest_roster.item_count):
        if guest_roster.get_item_metadata(row) == peer_id:
            guest_roster.remove_item(row)
            return

func _on_session_ended(_reason: StringName) -> void:
    guest_roster.clear()
```
