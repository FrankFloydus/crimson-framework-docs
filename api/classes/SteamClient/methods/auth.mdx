## Example

```gdscript
extends Node

const SIGN_IN_AUDIENCE: String = "game-profile"

# Assign the Steam client that your game already booted.
@export var steam_client: SteamClient

# From your sign-in flow, pass the Callable that sends bytes over your
# authenticated backend connection. It returns true after the backend
# verifies this audience and the game's AppID and accepts the player.
func sign_in(send_to_authenticated_backend: Callable) -> bool:
    if steam_client == null or not steam_client.is_ready():
        return false
    if not send_to_authenticated_backend.is_valid():
        return false
    var auth: SteamAuth = steam_client.auth()
    if auth == null:
        return false

    var issued: SteamWebTicketResult = await auth.web_ticket(SIGN_IN_AUDIENCE)
    if not issued.ok:
        return false
    var ticket: SteamWebTicket = issued.value
    if ticket == null:
        return false

    var proof: PackedByteArray = ticket.bytes()
    if proof.is_empty():
        ticket.close()
        return false
    var accepted: Variant = await send_to_authenticated_backend.call(proof)
    ticket.close()
    return accepted is bool and accepted
```
