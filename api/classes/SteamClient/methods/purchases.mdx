## Signature

```text
purchases() -> SteamPurchases
```

## Return value

[`SteamPurchases`](/api/classes/SteamPurchases)

## Description

A standalone client has no purchasing backend.

## Example

```gdscript
extends Node

@export var standalone_client: SteamClient
@export var game: GameRuntime
@export var shop_status: Label

# Inspect a standalone client booted without a GameRuntime or commerce adapter.
func show_standalone_shop_status() -> void:
    if standalone_client == null or shop_status == null:
        return
    var facade: SteamPurchases = standalone_client.purchases()
    if facade == null:
        shop_status.text = "Steam is unavailable."
        return
    var catalog: SteamPurchaseCatalogResult = await facade.catalog()
    if catalog.error == ERR_UNAVAILABLE:
        shop_status.text = "Start a game with an authenticated commerce backend."
        return
    if not catalog.ok:
        shop_status.text = catalog.message
        return
    shop_status.text = "%d offers for this player" % catalog.value.entries.size()

# Use a game already attached with SteamProvider.with_commerce_backend().
func refresh_attached_shop() -> void:
    if game == null or shop_status == null:
        return
    var scope: SteamScope = SteamServices.for_game(game)
    if scope == null or not scope.is_ready():
        shop_status.text = "Steam is unavailable for this game."
        return
    var shop: SteamPurchases = scope.purchases()
    if shop == null:
        shop_status.text = "The shop service is unavailable."
        return
    var catalog: SteamPurchaseCatalogResult = await shop.catalog()
    if not catalog.ok:
        shop_status.text = catalog.message
        return
    shop_status.text = "%d offers for this player" % catalog.value.entries.size()
```
