## Example

```gdscript
extends Node

func run_cloud(steam_client: SteamClient, total_bytes_label: Label) -> void:
    var result: SteamCloud = steam_client.cloud()
    if result == null:
        return
    var cloud_quota: SteamCloudQuotaResult = result.quota()
    if cloud_quota == null:
        return
    if not cloud_quota.ok:
        push_warning(cloud_quota.message)
        return
    var payload: SteamCloudQuota = cloud_quota.value
    if payload == null:
        return
    var total_bytes: int = payload.total_bytes
    total_bytes_label.text = "Total bytes: %s" % str(total_bytes)
```
