## Example

```gdscript
extends Node

func run_get_value(steam_identity_result: SteamIdentityResult, app_id_label: Label) -> void:
    if steam_identity_result == null:
        return
    if not steam_identity_result.ok:
        push_warning(steam_identity_result.message)
        return
    var result: SteamIdentity = steam_identity_result.get_value()
    if result == null:
        return
    var app_id: int = result.app_id
    app_id_label.text = "App id: %s" % str(app_id)
```
