## Example

```gdscript
extends Node

func refresh_value(steam_identity_result: SteamIdentityResult, app_id_label: Label) -> void:
    if steam_identity_result == null:
        return
    if not steam_identity_result.ok:
        push_warning(steam_identity_result.message)
        return
    var value: SteamIdentity = steam_identity_result.value
    if value == null:
        return
    var app_id: int = value.app_id
    app_id_label.text = "App id: %s" % str(app_id)
```
