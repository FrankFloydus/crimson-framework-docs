This backpack opens with a medkit and a radio. A drag keeps track of the cell you grabbed so an item can follow the pointer from any of its occupied cells. The excerpts below show how the complete script reads, checks, and commits a move.

## Record the grabbed cell

When the drag starts, convert the pointer position to a cell and look up its placement with [`item_at()`](/api/classes/SpatialInventory/methods/item_at). An empty cell returns `null`.

```gdscript
var cell: Vector2i = _cell_at(at_position)
var item: InventoryItemView = _bag.item_at(cell)
if item == null:
    return null

var drag: ItemDrag = ItemDrag.new()
drag.source = self
drag.id = item.id
drag.grab_offset = cell - item.origin
drag.rotated = item.rotated
```

Grabbing the radio's lower cell records an offset of `(0, 1)`. The ghost uses that offset while the item stays in its original cells.

## Check the destination

Subtract the grab offset from the hovered cell to find the proposed top-left cell. [`check_move()`](/api/classes/SpatialInventory/methods/check_move) checks the whole footprint at that position. The dragged item's own cells count as free, so the new placement can overlap its old one.

```gdscript
var origin: Vector2i = _cell_at(at_position) - drag.grab_offset
return _bag.check_move(drag.id, origin, drag.rotated) == OK
```

## Commit and redraw

On drop, [`try_move()`](/api/classes/SpatialInventory/methods/try_move) checks the destination again and commits the move if it still fits. On `OK`, redraw from the latest placements. An invalid drop leaves the original placement intact.

```gdscript
var origin: Vector2i = _cell_at(at_position) - drag.grab_offset
var error: Error = _bag.try_move(drag.id, origin, drag.rotated)
```

## Run the draggable backpack

Use a `Control` root with this complete script. It creates and draws the backpack.

```gdscript
extends Control

const BAG_SIZE: Vector2i = Vector2i(4, 3)
const CELL_PIXELS: int = 80
const GRID_ORIGIN: Vector2 = Vector2(24, 64)
const TITLE_POSITION: Vector2 = Vector2(24, 24)
const STATUS_GAP: float = 16.0
const ITEM_INSET: float = 3.0
const TEXT_INSET: Vector2 = Vector2(10, 25)
const MEDKIT_ID: StringName = &"medkit_17"
const MEDKIT_SIZE: Vector2i = Vector2i(2, 1)
const RADIO_ID: StringName = &"radio_02"
const RADIO_SIZE: Vector2i = Vector2i(1, 2)
const CELL_COLOR: Color = Color("#263546")
const GRID_COLOR: Color = Color("#526780")
const ITEM_COLOR: Color = Color("#29716d")
const GHOST_COLOR: Color = Color("#398e88aa")
const ITEM_NAMES: Dictionary[StringName, String] = {
    MEDKIT_ID: "Medkit",
    RADIO_ID: "Radio",
}

var _bag: SpatialInventory
var _status: Label
var _dragging: bool = false


class ItemDrag extends RefCounted:
    var source: Control
    var id: StringName
    var grab_offset: Vector2i
    var rotated: bool


func _ready() -> void:
    custom_minimum_size = GRID_ORIGIN + Vector2(BAG_SIZE * CELL_PIXELS) + Vector2(0, CELL_PIXELS)
    mouse_filter = Control.MOUSE_FILTER_STOP
    _add_label("Rearrange the backpack", TITLE_POSITION)
    var status_position: Vector2 = GRID_ORIGIN + Vector2(0, BAG_SIZE.y * CELL_PIXELS + STATUS_GAP)
    _status = _add_label("Drag an item to a free position.", status_position)

    _bag = Inventory.grid(BAG_SIZE)
    if _bag.add(MEDKIT_ID, MEDKIT_SIZE) != OK or _bag.add(RADIO_ID, RADIO_SIZE) != OK:
        _status.text = "The starter items did not fit."
        return
    queue_redraw()


func _add_label(caption: String, at: Vector2) -> Label:
    var label: Label = Label.new()
    label.text = caption
    label.position = at
    label.mouse_filter = Control.MOUSE_FILTER_IGNORE
    add_child(label)
    return label


func _draw() -> void:
    if _bag == null:
        return
    for row: int in range(BAG_SIZE.y):
        for column: int in range(BAG_SIZE.x):
            var cell: Rect2 = _rect_at(Vector2i(column, row), Vector2i.ONE)
            draw_rect(cell, CELL_COLOR)
            draw_rect(cell, GRID_COLOR, false)

    var font: Font = get_theme_default_font()
    for item: InventoryItemView in _bag.items():
        var bounds: Rect2 = _rect_at(item.origin, _occupied_size(item))
        draw_rect(bounds.grow(-ITEM_INSET), ITEM_COLOR)
        draw_string(font, bounds.position + TEXT_INSET, ITEM_NAMES[item.id])


func _get_drag_data(at_position: Vector2) -> Variant:
    var cell: Vector2i = _cell_at(at_position)
    var item: InventoryItemView = _bag.item_at(cell)
    if item == null:
        return null

    var drag: ItemDrag = ItemDrag.new()
    drag.source = self
    drag.id = item.id
    drag.grab_offset = cell - item.origin
    drag.rotated = item.rotated

    var preview: Control = Control.new()
    var ghost: ColorRect = ColorRect.new()
    ghost.color = GHOST_COLOR
    ghost.position = -Vector2(drag.grab_offset * CELL_PIXELS)
    ghost.size = Vector2(_occupied_size(item) * CELL_PIXELS)
    ghost.mouse_filter = Control.MOUSE_FILTER_IGNORE
    preview.add_child(ghost)
    set_drag_preview(preview)
    _status.text = "Moving %s." % ITEM_NAMES[item.id]
    _dragging = true
    return drag


func _can_drop_data(at_position: Vector2, data: Variant) -> bool:
    var drag: ItemDrag = data as ItemDrag
    if drag == null or drag.source != self:
        return false
    var origin: Vector2i = _cell_at(at_position) - drag.grab_offset
    return _bag.check_move(drag.id, origin, drag.rotated) == OK


func _drop_data(at_position: Vector2, data: Variant) -> void:
    var drag: ItemDrag = data as ItemDrag
    var origin: Vector2i = _cell_at(at_position) - drag.grab_offset
    var error: Error = _bag.try_move(drag.id, origin, drag.rotated)
    if error != OK:
        _status.text = "The item stayed in its original cells."
        return
    _status.text = "%s moved to %s." % [ITEM_NAMES[drag.id], origin]
    queue_redraw()


func _notification(what: int) -> void:
    if what == NOTIFICATION_DRAG_END and _dragging:
        _dragging = false
        if not get_viewport().gui_is_drag_successful():
            _status.text = "The item stayed in its original cells."


func _cell_at(point: Vector2) -> Vector2i:
    return Vector2i(((point - GRID_ORIGIN) / CELL_PIXELS).floor())


func _rect_at(origin: Vector2i, footprint: Vector2i) -> Rect2:
    return Rect2(GRID_ORIGIN + Vector2(origin * CELL_PIXELS), Vector2(footprint * CELL_PIXELS))


func _occupied_size(item: InventoryItemView) -> Vector2i:
    if item.rotated:
        return Vector2i(item.footprint.y, item.footprint.x)
    return item.footprint
```

Run the scene and drag the radio by its lower cell to the bottom-right cell. The radio's top-left cell moves to `(3, 1)`. Drag it back over the medkit or outside the grid to check that it stays put.

To add item quantities or choose a rotation, see [Items and stacks](/inventory/items). To move between separate grids, see [Inventory transactions](/inventory/transactions).
